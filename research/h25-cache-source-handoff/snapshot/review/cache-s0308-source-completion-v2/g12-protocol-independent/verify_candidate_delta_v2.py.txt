def read(path, expected):
    data=path.read_bytes(); digest=hashlib.sha256(data).hexdigest()
    assert digest == expected, (str(path),digest,expected)
    inputs[str(path)]={'sha256':digest,'bytes':len(data)}
    return data

def check(name, value):
    assert value,name
    checks.append(name)

def node(module,name):
    return next(n for n in module.body if isinstance(n,(ast.FunctionDef,ast.ClassDef)) and n.name==name)

def own_functions(module):
    return {n.name:n for n in module.body if isinstance(n,ast.FunctionDef)}

def bound_names(function):
    names={n.id for n in ast.walk(function) if isinstance(n,ast.Name) and isinstance(n.ctx,(ast.Store,ast.Del))}
    names.update(n.arg for n in ast.walk(function.args) if isinstance(n,ast.arg))
    names.update(n.name for n in ast.walk(function) if n is not function and isinstance(n,(ast.FunctionDef,ast.ClassDef)))
    names.update(n.name for n in ast.walk(function) if isinstance(n,ast.ExceptHandler) and n.name is not None)
    return names
