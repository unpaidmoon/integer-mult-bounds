def sha(data):
    return hashlib.sha256(data).hexdigest()


def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"), ensure_ascii=True).encode("ascii")


def cpu():
    own = resource.getrusage(resource.RUSAGE_SELF)
    children = resource.getrusage(resource.RUSAGE_CHILDREN)
    return own.ru_utime + own.ru_stime + children.ru_utime + children.ru_stime


def inventory(root, pins, *, exact=True):
    need(root.is_dir() and not root.is_symlink(), "Missing or symlinked input root")
    files = {}
    for path in sorted(root.rglob("*")):
        need(not path.is_symlink(), "Symlink in frozen inputs: " + str(path))
        if path.is_file():
            relative = str(path.relative_to(root))
            if relative not in pins:
                need(not exact, "Undeclared frozen input: " + relative)
                continue
            raw = path.read_bytes()
            expected = pins[relative][0] if type(pins[relative]) is list else pins[relative]
            need(sha(raw) == expected, "Frozen input digest changed: " + relative)
            files[relative] = {"sha256": sha(raw), "mode": stat.S_IMODE(path.stat().st_mode), "bytes": len(raw)}
    need(set(files) == set(pins), "Frozen input inventory incomplete")
    return files

def typed(value):
    """Lossless full return representation, including order and exact types."""
    kind = type(value)
    if kind is Fraction:
        return ["Fraction", value.numerator, value.denominator]
    if value is None:
        return ["None"]
    if kind is bool:
        return ["bool", value]
    if kind is int:
        return ["int", str(value)]
    if kind is float:
        return ["float", value.hex()]
    if kind is str:
        return ["str", value]
    if kind is list or kind is tuple:
        return [kind.__name__, [typed(item) for item in value]]
    if kind is dict:
        return ["dict", [[typed(key), typed(item)] for key, item in value.items()]]
    raise TypeError("Unsupported exact full-certificate return type: " + kind.__name__)


class Phases:
    """Broad phase observations; no per-arithmetic resource sampling.

    Every call retains the old timing fields. Memory samples are paid by both
    arms, including nested side_construction so snapshots/retirement are visible.
    """
    def __init__(self):
        self.rows = {}

    @staticmethod
    def memory():
        with open("/proc/self/status", "rb") as stream:
            raw = stream.read(65537)
        need(len(raw) <= 65536, "Process status diagnostic exceeded its bound")
        current = None
        for line in raw.splitlines():
            if line.startswith(b"VmRSS:"):
                fields = line.split()
                need(len(fields) == 3 and fields[2] == b"kB", "Unexpected process RSS diagnostic units")
                current = int(fields[1]) * 1024
                break
        need(current is not None, "Current process RSS diagnostic unavailable")
        highwater = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss * 1024
        return {"current_vmrss_bytes": current, "process_ru_maxrss_bytes": highwater}

    def call(self, name, function, *args, **kwargs):
        wall, used = time.perf_counter(), cpu()
        memory_before = self.memory()
        primary_failure = False
        try:
            return _g12_after_phase(self, name, function, args, kwargs, function(*args, **kwargs))
        except BaseException:
            primary_failure = True
            raise
        finally:
            memory_after, memory_error = None, None
            try:
                memory_after = self.memory()
            except BaseException as error:
                # Do not replace an original failure with diagnostic failure.
                memory_error = {"type": type(error).__name__}
                if not primary_failure:
                    raise
            row = self.rows.setdefault(name, {"calls": 0, "wall_seconds": 0.0,
                                             "cpu_seconds_with_children": 0.0,
                                             "memory_samples": [],
                                             "memory_semantics": "VmRSS is current resident bytes; ru_maxrss is process high-water bytes; neither is an additive phase peak"})
            row["calls"] += 1
            row["memory_samples"].append({"before": memory_before, "after": memory_after,
                                           "diagnostic_error": memory_error})
            row["wall_seconds"] += time.perf_counter() - wall
            row["cpu_seconds_with_children"] += cpu() - used

    def wrap(self, owner, attribute, name):
        original = getattr(owner, attribute)
        def observed(*args, **kwargs):
            return self.call(name, original, *args, **kwargs)
        setattr(owner, attribute, observed)
        return original, observed

def _run_with_stop_evidence():
    try:
        main()
    except BaseException as primary_error:
        # Nothing in cleanup or evidence assembly may replace this primary object.
        try:
            state = _EXECUTION['state']
            cleanup_errors = []
            try:
                cleanup_errors = data_adapters.cleanup_owned(state)
            except BaseException as cleanup_error:
                cleanup_errors.append({'type': type(cleanup_error).__name__, 'message': str(cleanup_error)})
            stop = {'schema': 'h25-combined-v2-stop-v1', 'status': 'STOP',
                    'primary_error': {'type': type(primary_error).__name__, 'message': str(primary_error)},
                    'primary_cause': None if primary_error.__cause__ is None else {
                        'type': type(primary_error.__cause__).__name__, 'message': str(primary_error.__cause__)},
                    'cleanup_errors': cleanup_errors,
                    'data': data_adapters.stop_diagnostics(state),
                    'native_packet': None if _EXECUTION['packet'] is None else native_packet.receipt(_EXECUTION['packet']),
                    'initial_native_tcb': INITIAL_NATIVE_TCB_RECEIPT}
            output = _EXECUTION['output']
            if output is not None:
                with (output / 'STOP.json').open('x', encoding='utf-8') as stream:
                    stream.write(json.dumps(stop, sort_keys=True, indent=2) + '\n')
        except BaseException as evidence_error:
            # Parent retains an independent process transcript even if this fails.
            try:
                sys.stderr.write('STOP evidence write failed: ' + type(evidence_error).__name__ + '\n')
            except BaseException:
                pass
        raise




def _epoch_cli_boundary(phases, data_state, extensions, caller, stdout, invocation_token, old_argv):
    _epoch_cli_primary = None
    _epoch_cli_argv_error = None
    _epoch_cli_disarm_error = None
    _epoch_cli_retirement_error = None

            with redirect_stdout(stdout):
                namespace = phases.call('complete_original_cli', runpy.run_path, str(caller), run_name='__main__')

        except BaseException as _epoch_cli_error:
            _epoch_cli_primary = _epoch_cli_error
            if _epoch_cli_record is not None:
                _epoch_cli_record['primary'] = _epoch_cli_error
            raise
        finally:
            try:
                sys.argv = old_argv
            except BaseException as _epoch_argv_error:
                _epoch_cli_argv_error = _epoch_argv_error
                if _epoch_cli_record is not None:
                    _epoch_cli_record['argv_error'] = _epoch_argv_error
                    _epoch_cli_record['uncertain'] = True

            try:

                _epoch_cli_disarm_error = _epoch_disarm_error
                if _epoch_cli_record is not None:
                    _epoch_cli_record['disarm_error'] = _epoch_disarm_error
                    _epoch_cli_record['uncertain'] = True
                if _epoch_cli_primary is None:
                    _epoch_cli_primary = _epoch_disarm_error
                    if _epoch_cli_record is not None:
                        _epoch_cli_record['primary'] = _epoch_disarm_error
                    raise
            if _epoch_cli_argv_error is not None:
                raise _epoch_cli_primary
    finally:
        try:
            data_adapters.epoch_retire(data_state, 'whole-cli-terminal', _epoch_cli_primary, _epoch_cli_disarm_error, _epoch_cli_argv_error, None if _epoch_cli_record is None else _epoch_cli_record['retirement_channel'], 'inner')
        except BaseException as _epoch_retirement_error:
            _epoch_cli_retirement_error = _epoch_retirement_error
            if _epoch_cli_record is not None:
                _epoch_cli_record['retirement_error'] = _epoch_retirement_error
                _epoch_cli_record['uncertain'] = True
            if _epoch_cli_primary is None:
                _epoch_cli_primary = _epoch_retirement_error
                if _epoch_cli_record is not None:
                    _epoch_cli_record['primary'] = _epoch_retirement_error
                raise

def _g12_primitive(value):
    """Lossless tagged exact builtin structures; never invoke foreign equality."""
    kind = type(value)
    if kind is Fraction:
        return ['Fraction', str(value.numerator), str(value.denominator)]
    if value is None:
        return ['None']
    if kind is bool:
        return ['bool', value]
    if kind is int:
        return ['int', str(value)]
    if kind is str:
        return ['str', value]
    if kind is bytes:
        return ['bytes', value.hex()]
    if kind is float:
        return ['float', value.hex()]
    if kind is tuple or kind is list:
        return [kind.__name__, [_g12_primitive(item) for item in value]]
    if kind is dict:
        return ['dict', [[_g12_primitive(key), _g12_primitive(item)]
                         for key, item in value.items()]]
    raise TypeError('G12 common projection requires exact original primitive types')


def _g12_bytes(value):
    return canonical(_g12_primitive(value))

            except StopIteration as early:
                raise RuntimeError('G12 P07 completed before original CLI remainder') from early
            record['observed'] = observed
            _g12_generator_shape(True)
            record['step'] = 'P07_SUSPENDED'
        else:
            need(_G12_SELECTED_CASE == 'N22', 'G12 source selected an undeclared helper')
            _g12_helper_current()
            record['observed'] = record['helper_function'](context)
            record['step'] = 'N22_RETURNED'
        need(type(record['observed']) is dict and
             all(type(key) is str for key in record['observed']) and
             set(record['observed']) == {'value', 'capacity_retirements', 'owner'} and
             suite.terminal is True and suite.active is False and not suite.registry and
             suite.pending is None and suite.request_live is None,
             'G12 original N22 did not return its exact terminal continuation observation')
        # N22 performed the sole original commit. Never call it again here.
        return None
    except BaseException as primary:
        if record['primary'] is None:
            record['primary'] = primary
        raise

def _g12_after_phase(phases, name, function, args, kwargs, result):
    record = _G12_CONTINUATION
    if not record['ready']:
        return result
    if name not in ('compile_roles', 'verify_labels', 'complete_original_cli'):
        if record['step'] == 'SUFFIX' and name in _G12_SUFFIX_PHASES:
            record['suffix_phases'].append(name)
        return result
    need(phases is record['phases'], 'G12 fixed actual phase owner changed')
    _g12_helper_current()
    if name == 'compile_roles':
        need(record['step'] == 'CLI_READY' and type(args) is tuple and len(args) == 1 and not kwargs and
             function is record['code_pair'][0] and record['module'].compile_roles is record['code_pair'][1] and
             type(args[0]) is record['module'].ComplexSideCircuit,
             'G12 actual original compile-role return checkpoint changed')
        record.update({'circuit': args[0], 'code': result, 'code_before': _g12_code_snapshot(result), 'step': 'COMPILED'})
        return result
    if name == 'verify_labels':
        need(record['step'] == 'COMPILED' and type(args) is tuple and len(args) == 1 and not kwargs and
             function is record['label_pair'][0] and record['module'].Checks.__dict__['verify_labels'] is record['label_pair'][1],
             'G12 actual original label-phase return checkpoint changed')
        suite, checks, circuit = record['suite'], args[0], record['circuit']
        set_owner = record['extensions']['set_binding']['owner']
        need(type(checks) is suite.checks_class and checks.c is circuit and
             type(checks.h) is int and type(circuit.h) is int and checks.h == circuit.h == 25 and
             suite.active is True and suite.terminal is False and suite.invocation_consumed is True and
             suite.primary_error is None and suite.pending is None and suite.request_live is None and
             suite.checks is checks and suite.checks_c is circuit and
             set_owner.cli_active is True and set_owner.used is False and set_owner.circuit is circuit and
             set_owner.checks is checks and set_owner.triples is circuit.triples and
             type(suite.registry) is dict and len(suite.registry) == 85196 and type(checks.cache) is dict and
             len(checks.cache) == 85195 and ('perp', (0, 2, 1)) not in checks.cache,
             'G12 genuine post-label original capacity checkpoint changed')
        _g12_complete_capacity_store(checks, circuit, suite)
        cleanup = record['state']['epoch_cli_cleanup']
        need(type(cleanup) is dict and cleanup['primary'] is None and cleanup['argv_error'] is None and
             cleanup['disarm_error'] is None and cleanup['retirement_error'] is None and
             type(sys.argv) is list and sys.argv == [record['caller']] and sys.argv is not cleanup['argv_owner'],
             'G12 must be inside original protected CLI after actual arming')
        c1 = suite.registry.get(id(checks.everything))
        need(type(c1) is tuple and len(c1) == 7 and c1[0] is checks.everything and c1[1] is checks.everything.basis and
             c1[2] is checks.everything.key and c1[4] is checks and c1[5] == 'C1' and c1[6] == suite.generation,
             'G12 registered_label must be actual original C1 store')
        need(_g12_code_snapshot(record['code']) == record['code_before'], 'G12 original code changed before control store')
        literal, reference_count, reference_outputs, _ = record['reference']
        record['checks'] = checks
        record['context'] = {'bundle': _INITIAL_EPOCH_BUNDLE, 'state': record['state'], 'site': 'C3',
                             'checks': checks, 'registered_label': checks.everything, 'literal_expected': literal,
                             'reference_remaining_case_count': reference_count, 'reference_remaining_outputs': reference_outputs}
        record['prefix_end'] = _g12_diagnostics()
        record['step'] = 'EXPECT_STORE'
        actual_label = checks.target_frame((0, 2, 1))
        need(record['store_entries'] == 1 and actual_label is record['stored_label'] and
             checks.cache[('perp', (0, 2, 1))] is actual_label and
             record['step'] == ('P07_SUSPENDED' if _G12_SELECTED_CASE == 'P07' else 'N22_RETURNED'),
             'G12 actual selected store did not complete exactly once')
        record['suffix_start'] = _g12_diagnostics()
        record['rank_prefix'] = _g12_bytes(tuple(record['rank_calls']))
        record['rank_index'] = len(record['rank_calls'])
        record['step'] = 'SUFFIX'
        return result
    need(record['step'] == 'SUFFIX' and function is runpy.run_path and type(args) is tuple and len(args) == 1 and
         args[0] == record['caller'] and type(kwargs) is dict and tuple(kwargs) == ('run_name',) and kwargs['run_name'] == '__main__',
         'G12 actual same-invocation original CLI return changed')
    record['suffix_phases'].append(name)
    _g12_complete_original_cli(result)
    marker = _G12ControlCompletion('G12 intentional ' + _G12_SELECTED_CASE + ' original-CLI control completion')
    record['marker'], record['step'] = marker, 'MARKED'
    raise marker

def _g12_after_original_exit(primary):
    record = _G12_CONTINUATION
    if record['primary'] is None:
        record['primary'] = primary
    generator = record['generator']
    if generator is not None and generator.gi_frame is not None:
        try:
            # No original CLI resume occurs on failure. Original P07 source has
            # no finally clause; closing this exact suspended code only releases
            # its retained frame after original CLI/main/STOP/owned cleanup.
            _g12_generator_shape(True)
            need(record['close_attempts'] == 0, 'G12 suspended P07 close repeated')
            record['close_attempts'] = 1
            generator.close()
            need(generator.gi_frame is None, 'G12 exact suspended P07 did not close')
        except BaseException as secondary:
            try:
                record['secondary_clear'] = False
                record['frame_close_error'] = secondary
                record['secondary'].append((secondary, secondary.__traceback__, 'P07-frame-close'))
            except BaseException:
                pass
    if primary is record['marker'] and record['step'] == 'MARKED':
        try:
            _g12_original_terminal_custody()
            need(record['secondary_clear'] is True and
                 record['frame_close_error'] is None and record['terminal_observation_error'] is None and
                 record['adapter_exit_error'] is None and not record['secondary'],
                 'G12 source-control secondary error or incomplete journal blocks observation export')
            row = {'schema': 'g12-original-control-observation-v1', 'case': _G12_SELECTED_CASE,
                   'selected_driver_sha256': sha(record['driver_before']),
                   'reference_provenance': record['reference'][3],
                   'original_helper_observation': _g12_primitive(record['observed']),
                   'remaining_original_case_count': record['case_count'],
                   'common_payload_sha256': sha(record['common']), 'common_payload_hex': record['common'].hex(),
                   'prefix_end': record['prefix_end'], 'suffix_start': record['suffix_start'], 'cli_end': record['cli_end'],
                   'ordered_rank_suffix_hex': record['rank_suffix'].hex(), 'native_suffix_delta': record['native_suffix_delta'],
                   'phases': record['phases'].rows,
                   'store_entries': record['store_entries'], 'first_advances': record['first_advances'],
                   'second_advances': record['second_advances'], 'close_attempts': record['close_attempts'],
                   'disposition_qualification': 'OPEN_REQUIRES_SEPARATE_ORIGINAL_STOP_AND_ALL_OWNED_CLEANUP_REVIEW',
                   'scope': 'Original selected N22/P07 observation plus same-invocation original CLI remainder only. No original full-positive publication/native-count/P02/G12/FULL admission.'}
            with (record['output'] / 'g12-control-observation.json').open('xb') as stream:
                stream.write(canonical(row) + b'\n')
        except BaseException as secondary:
            try:
                record['secondary_clear'] = False
                record['terminal_observation_error'] = secondary
                record['secondary'].append((secondary, secondary.__traceback__, 'terminal-source-observation'))
            except BaseException:
                pass
    for secondary, traceback, phase in record['secondary']:
        try:
            sys.stderr.write('G12 secondary continuation error: ' + phase + '\n')
            _g12_traceback.print_exception(type(secondary), secondary, traceback, file=sys.stderr)
        except BaseException:
            pass


def _g12_continuation_entry():
    record = _G12_CONTINUATION
    need(record['started'] is False and _G12_SELECTED_CASE in ('N22', 'P07'), 'G12 source control entry is one-use')
    record['started'] = True
    record['driver'] = sys.modules['__main__']
    need(record['driver'].__dict__ is globals(), 'G12 source control requires actual Root-installed driver namespace')
    need(type(_G12_TRANSPORT_SOURCE_PIN) is str and len(_G12_TRANSPORT_SOURCE_PIN) == 64,
         'G12 reference transport source has not been selected')
    helper, helper_handle = _g12_cold(0, str(ROUND / 'areas/cache-validity/g12_source_controls.py'),
                                     _G12_SELECTED_HELPER_PIN, '_h25_g12_source_controls')
    transport, transport_handle = _g12_cold(0, str(ROUND / 'areas/cache-validity/g12_reference_transport.py'),
                                           _G12_TRANSPORT_SOURCE_PIN, '_h25_g12_reference_transport')
    reference = _g12_cold(1, transport_handle, 'load_original_g12_reference',
                          _CAPTURED_BOOTSTRAP_ROOT, _CAPTURED_BOOTSTRAP_API)
    need(type(reference) is tuple and len(reference) == 4 and type(reference[0]) is bool and
         type(reference[1]) is int and reference[1] > 0 and type(reference[2]) is bytes and type(reference[3]) is tuple,
         'G12 actual source-produced independent reference transport shape changed')
    source = _g12_cold(2, helper_handle)
    function = helper.__dict__[_G12_SELECTED_CASE]
    need(source[1] is helper and source[5] == _G12_SELECTED_HELPER_PIN and
         type(function) is _g12_types.FunctionType and function.__globals__ is helper.__dict__ and
         bool(function.__code__.co_flags & 0x20) is (_G12_SELECTED_CASE == 'P07'),
         'G12 exact selected helper source/generator protocol changed')
    record.update({'helper_handle': helper_handle, 'helper_module': helper, 'helper_function': function,
                   'helper_code': function.__code__, 'transport_handle': transport_handle,
                   'reference': reference, 'generator_type': _g12_types.GeneratorType, 'step': 'PREPARED'})
    try:
        _run_epoch_with_stop_evidence()
    except BaseException as primary:
        # Original CLI/main/STOP/owned cleanup has already run. Preserve the
        # identical primary and cause; every adapter cleanup error is secondary.
        try:
            _g12_after_original_exit(primary)
        except BaseException as secondary:
            try:
                record['secondary_clear'] = False
                record['adapter_exit_error'] = secondary
                record['secondary'].append((secondary, secondary.__traceback__, 'adapter-exit-entry'))
            except BaseException:
                pass
        raise
    raise RuntimeError('G12 source control reached original full-positive return without its intentional marker')
