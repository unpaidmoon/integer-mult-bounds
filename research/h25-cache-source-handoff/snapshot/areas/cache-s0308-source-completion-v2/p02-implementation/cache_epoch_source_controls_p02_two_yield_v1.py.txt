          p[0] is type and p[1] is types.FunctionType and p[2] is id and p[18] is zip,
          'Control genuine native caller primitives')
    _need(type(fn) is p[1] and fn is r[0],'Control actual dispatcher')
    for (_,get),value in p[18](p[4],r[1:7]):
        _need(get(fn,p[1]) is value,'Control dispatcher fields')

    for cell,value in r[8]:_need(p[6](cell,p[5]) is value,'Control dispatcher cell')
