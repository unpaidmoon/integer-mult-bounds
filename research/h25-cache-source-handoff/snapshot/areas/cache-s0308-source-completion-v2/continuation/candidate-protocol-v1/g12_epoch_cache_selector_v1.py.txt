"""Source-derived verdict wrappers with current-data and live-runtime authority.

Parent-owned installation only. No scientific module imports, arithmetic
replacement, immutable Label replacement or shared reference answers. Generic
Label data may use literal uncached expressions; custom Checks/verdict mappings
are rejected. Serial stability is the existing native backend contract.

V7 preserves the V5 public-dictionary collision contract, the V6 backend/deep
guard split, and the real row interner at both full and light guard boundaries.
All cold entries are invoked through one source-closed early authority. Ordered
base/trace preflight retains the original existing-drift first error. Private
owner/guard state corruption remains excluded; no instance-key safety is claimed.
This source-only draft requires independent source and concrete stage admission.

V5 replaces the all-public entry census with a fresh admitted native collision-
path inspector. The inspector returns True or NotImplemented metadata only;
entry and exit always inspect current state. All source arithmetic/public dict
operations, canonical scans and authority remain Python-owned and unchanged.
No path, version stamp, validation PASS or arithmetic answer is saved by the
inspector. Compile/runtime/full gates remain closed pending independent review.

The concrete caller must exclude unrelated callbacks/finalizers, signals,
pending callbacks, concurrent mutation and private corruption for the whole
admitted request: initial guard through key/preparation/probes/original work,
exit guards and authority commit/return. GIL ownership and snapshots alone do
not establish that premise. Unsupported fallback retains separately scoped
literal callbacks and receives no authority.

"""

def _need(condition, message):
    if not condition:
        raise _ERROR(message)


def _record(function):
    _need(_TYPE(function) is FunctionType, "Cache contract requires ordinary functions")
    closure = function.__closure__
    return (function, function.__code__, function.__globals__, function.__builtins__,
            function.__defaults__, function.__kwdefaults__, closure,
            () if closure is None else _TUPLE((cell, cell.cell_contents) for cell in closure))


def _fresh(record):
    function, code, namespace, builtins_owner, defaults, kwdefaults, closure, cells = record
    if not (function.__code__ is code and function.__globals__ is namespace and
            function.__builtins__ is builtins_owner and function.__defaults__ is defaults and
            function.__kwdefaults__ is kwdefaults and function.__closure__ is closure):
        return False
    for cell, value in cells:
        try:
            if cell.cell_contents is not value:
                return False
        except _VALUE_ERROR:
            return False
    return True


def _string_key_domain(namespace):
    """Return shape metadata only; release observed key/iterator before return.

    Exact dict does not by itself prevent equality callbacks on named lookup.
    The closed guard interval excludes concurrent mutation and unrelated
    allocation finalizers; this function does not disable garbage collection.
    """
    if _TYPE(namespace) is not _DICT:
        return False
    current = iterator = None
    try:
        iterator = _DICT.__iter__(namespace)
        for current in iterator:
            if _TYPE(current) is not _STR:
                return False
        return True
    finally:
        current = None
        iterator = None
        namespace = None


def _observed_fresh(record, observations):
    """Share scalar actual metadata, never a PASS or independently expected row.

    Function/cell objects below come from retained expected records. Current
    defaults/code/globals/cell values are consumed only by captured id and are
    never bound to local variables or retained in a traceback. Every expected
    record still compares separately, in original field/cell order.
    """
    current = key = None
    try:
        function, code, namespace, builtins_owner, defaults, kwdefaults, closure, cells = record
        if _TYPE(function) is not FunctionType:
            return False
        key = (0, _ID(function))
        current = _GET(observations, key, _MISSING)
        if current is _MISSING:
            current = (_ID(function.__code__), _ID(function.__globals__),
                       _ID(function.__builtins__), _ID(function.__defaults__),
                       _ID(function.__kwdefaults__), _ID(function.__closure__))
            observations[key] = current
        if not (current[0] == _ID(code) and current[1] == _ID(namespace) and
                current[2] == _ID(builtins_owner) and current[3] == _ID(defaults) and
                current[4] == _ID(kwdefaults) and current[5] == _ID(closure)):
            return False
        for cell, value in cells:
            key = (1, _ID(cell))
            current = _GET(observations, key, _MISSING)
            if current is _MISSING:
                try:
                    current = (True, _ID(cell.cell_contents))
                except _VALUE_ERROR:
                    current = (False, 0)
                observations[key] = current
            if not current[0] or current[1] != _ID(value):
                return False
        return True
    finally:
        current = key = None
        observations = None


def _ordinary(cls):
    if _TYPE(cls) is not _TYPE or _TYPEGET(cls, "__mro__") != (cls, _OBJECT):
        return False
    members = _TYPEGET(cls, "__dict__")
    if "__getattr__" in members or _TYPEGET(cls, "__getattribute__") is not _OBJECT.__getattribute__:
        return False
    if _TYPEGET(cls, "__new__") is not _OBJECT.__new__:
        return False
    for name in ("__dict__", "__weakref__"):
        descriptor = members.get(name)
        if _TYPE(descriptor) is not GetSetDescriptorType or descriptor.__objclass__ is not cls:
            return False
    return True


def _plain_fields(instance, cls):
    if _TYPE(instance) is not cls:
        return None
    namespace = _OBJGET(instance, "__dict__")
    if _TYPE(namespace) is not _DICT:
        return None
    # Validate before string-key lookup: injected custom keys could otherwise
    # invoke equality against an ordinary field name.
    for name in namespace:
        if _TYPE(name) is not _STR:
            return None
    return namespace


def _canonical(instance, label):
    """Current exact data, never Label.key alone; no custom operand dispatch."""
    namespace = _plain_fields(instance, label)
    if namespace is None or _LEN(namespace) != 3:
        return None
    basis = _GET(namespace, "basis", _MISSING)
    dim = _GET(namespace, "dim", _MISSING)
    key = _GET(namespace, "key", _MISSING)
    if _TYPE(basis) is not _DICT or _TYPE(dim) is not _INT or _TYPE(key) is not _TUPLE:
        return None
    if dim != _LEN(basis) or _LEN(key) != dim:
        return None
    for value in key:
        if _TYPE(value) is not _INT:
            return None
    pivots = 0
    values = []
    for pivot, value in _ITEMS(basis):
        if _TYPE(pivot) is not _INT or _TYPE(value) is not _INT:
            return None
        if pivot <= 0 or pivot & (pivot - 1) or value <= 0 or value & -value != pivot:
            return None
        pivots |= pivot
        values.append(value)
    for pivot, value in _ITEMS(basis):
        if value & pivots != pivot:
            return None
    if _TUPLE(_SORTED(values)) != key:
        return None
    # key is now an exact immutable representation of the canonical mapping.
    return key


def _hooks_clear():
    """Known hook exclusion, not a promise against signals/finalizers/threads."""
    if _GETTRACE() is not None or _GETPROFILE() is not None or _THREADCOUNT() != 0:
        return False
    callbacks = gc.callbacks
    if _TYPE(callbacks) is not list or _LEN(callbacks):
        return False
    monitoring = getattr(sys, "monitoring", None)
    if monitoring is not None:
        for tool in range(6):
            if monitoring.get_tool(tool) is not None:
                return False
    return True

    def _clear(self, state):
        count = _LEN(state.authority)
        if count:
            _CLEAR(state.authority)
            self.receipt["current_authority_entries"] -= count
            self.receipt["invalidations"] += 1

    def _lookup_state(self, checks):
        state = _GET(self.states, _ID(checks))
        if state is not None and state.checks is checks:
            return state
        return None

    def enter(self, checks):
        _need(self.runtime is not None, "Cache contract runtime is not attached")
        self.runtime.check()
        self.receipt["entry_checks"] += 1
        self.receipt["requests"] += 1
        # Do not read verdicts/everything/key here. Original key errors precede
        # original verdict access errors; custom Checks itself is unsupported.
        _need(_plain_fields(checks, self.checks_class) is not None,
              "Unsupported cache contract Checks dispatch or fields")
        return checks

    def _data(self, checks, key, operands, kind):
        self.receipt["snapshots"] += 1
        namespace = _plain_fields(checks, self.checks_class)
        if namespace is None:
            return None
        circuit, height = _GET(namespace, "c", _MISSING), _GET(namespace, "h", _MISSING)
        if circuit is _MISSING or _TYPE(height) is not _INT:
            return None
        keys = []
        for operand in operands:
            value = _canonical(operand, self.label)
            if value is None:
                return None
            keys.append(value)
        if kind == "label":
            everything = _GET(namespace, "everything", _MISSING)
            whole_key = _canonical(everything, self.label)
            if whole_key is None or _TYPE(key) is not _TUPLE or key != ("label", keys[0]):
                return None
            signature = (kind, keys[0], whole_key)
        else:
            if _TYPE(key) is not _TUPLE or key != (keys[0], keys[1]):
                return None
            signature = (kind, keys[0], keys[1])
        return signature, circuit, height

    def _probe_safe(self, mapping, key, *, exit_boundary=False):
        """Fresh metadata only. inspect itself checks admitted provider/native state."""
        self.check_bootstrap()
        counter = "probe_exit_calls" if exit_boundary else "probe_entry_calls"
        self.receipt[counter] += 1
        result = self.probe_inspect(mapping, key)
        if result is True:
            self.receipt["probe_safe_observations"] += 1
            return True
        if result is _NOT_IMPLEMENTED:
            self.receipt["probe_unsupported_observations"] += 1
            return False
        raise _ERROR("Cache probe returned non-metadata result")

    def prepare(self, checks, key, operands, kind):
        namespace = _plain_fields(checks, self.checks_class)
        _need(namespace is not None, "Unsupported cache contract Checks fields")
        mapping = _GET(namespace, "verdicts", _MISSING)
        state = self._lookup_state(checks)
        if mapping is _MISSING:
            if state is not None:
                self._clear(state)
            self.receipt["uncached_requests"] += 1
            return None  # original `self.verdicts` raises at its original site
        _need(_TYPE(mapping) is _DICT, "Unsupported cache contract verdict mapping")
        if state is None or state.mapping is not mapping:
            if state is not None:
                self._clear(state)
            state = _State(checks, mapping)
            self.states[_ID(checks)] = state
        # Preserve original key construction and canonical/type admission before
        # the native helper may hash an exact supported request key.
        data = self._data(checks, key, operands, kind) if _hooks_clear() else None
        public_safe = data is not None and self._probe_safe(mapping, key)
        if not public_safe:
            self._clear(state)  # no hash/equality involving unsupported key
            self.receipt["uncached_requests"] += 1
            return None
        self.receipt["eligible_requests"] += 1
        signature, circuit, height = data
        return _Ticket(state, signature, circuit, height)

    def reuse(self, ticket, key):
        if ticket is None:
            return False
        record = _GET(ticket.state.authority, key)
        if record is None:
            return False
        signature, circuit, height, result = record
        if signature != ticket.signature or circuit is not ticket.c or height != ticket.h:
            return False
        ticket.hit, ticket.expected = True, result
        return True

    def finish(self, checks, key, operands, kind, ticket, result):
        self.runtime.check()
        self.receipt["exit_checks"] += 1
        if ticket is None:
            state = self._lookup_state(checks)
            if state is not None:
                self._clear(state)  # callbacks may have made reentrant requests
            return result
        namespace = _plain_fields(checks, self.checks_class)
        same_mapping = namespace is not None and _GET(namespace, "verdicts", _MISSING) is ticket.state.mapping
        data = self._data(checks, key, operands, kind) if same_mapping and _hooks_clear() else None
        if (not same_mapping or
                data is None or data[0] != ticket.signature or
                data[1] is not ticket.c or data[2] != ticket.h):
            self._clear(ticket.state)
            raise _ERROR("Cache data changed during request")
        if not self._probe_safe(ticket.state.mapping, key, exit_boundary=True):
            self._clear(ticket.state)
            self.receipt["probe_exit_rejections"] += 1
            raise _ERROR("Cache probe unsupported at exit")
        if ticket.hit and result is not ticket.expected:
            self._clear(ticket.state)
            raise _ERROR("Cache verdict authority changed")
        if _TYPE(result) is not _BOOL:
            self._clear(ticket.state)
            return result
        if ticket.hit:
            self.receipt["authorized_hits"] += 1
        authority = ticket.state.authority
        if key not in authority:
            self.receipt["current_authority_entries"] += 1
            count = self.receipt["current_authority_entries"]
            if count > self.receipt["peak_authority_entries"]:
                self.receipt["peak_authority_entries"] = count
        authority[key] = (ticket.signature, ticket.c, ticket.h, result)
        self.receipt["authorized_results"] += 1
        return result

    def abort(self, checks):
        state = self._lookup_state(checks)
        if state is not None:
            self._clear(state)

    def teardown(self):
        for state in self.states.values():
            self._clear(state)
        _CLEAR(self.states)
        self.receipt["teardowns"] += 1
        _need(not self.probe_closed, "Cache probe was already closed")
        expected_calls = self.receipt["probe_entry_calls"] + self.receipt["probe_exit_calls"]
        native_counters = None
        primary_error = close_error = None
        stage = "validation"
        try:
            self.check_bootstrap()
            self.probe_validate(self.probe_binding)
            stage = "counters"
            native_counters = self.probe_counters()
            stage = "accounting"
            _need(_TYPE(native_counters) is _DICT and
                  _TYPE(_GET(native_counters, "calls")) is _INT and
                  native_counters["calls"] == expected_calls and
                  _TYPE(_GET(native_counters, "counter_overflow")) is _INT and
                  native_counters["counter_overflow"] == 0,
                  "Cache native probe call accounting or saturation differs")
        except _BASE_ERROR as error:
            primary_error = error
        finally:
            try:
                # Captured provider close attempts its captured native disarm and
                # registry removal even when its own freshness/audit rejects.
                self.probe_close()
            except _BASE_ERROR as error:
                close_error = error
            finally:
                self.probe_closed = True
        if primary_error is not None or close_error is not None:
            self.teardown_primary_error = primary_error
            self.teardown_close_error = close_error
            observed = (_GET(native_counters, "calls")
                        if _TYPE(native_counters) is _DICT else None)
            self.teardown_stop = {
                "status": "STOP", "stage": stage if primary_error is not None else "close",
                "expected_probe_calls": expected_calls,
                "observed_probe_calls": observed,
                "attempted_minus_native": (expected_calls - observed
                                           if _TYPE(observed) is _INT else None),
                "native_probe": native_counters,
                "native_counters_available": native_counters is not None,
                "close_attempted": True, "owner_terminal": True,
                "primary_error": (None if primary_error is None else
                                  (_TYPE(primary_error).__name__, _STR(primary_error))),
                "close_error": (None if close_error is None else
                                (_TYPE(close_error).__name__, _STR(close_error))),
            }
            # Preserve the actual primary exception object, including its
            # traceback. A cleanup failure remains explicit as its cause.
            if primary_error is not None:
                if close_error is not None:
                    raise primary_error from close_error
                raise primary_error
            raise close_error
        result = _DICT(self.receipt)
        result["native_probe"] = native_counters
        result["probe_closed"] = True
        return result

        if _TYPE(item) is CodeType and item.co_name == name:

          function.__builtins__ is _BUILTINS and function.__closure__ is None and

def _generate(module, original_ast, owner, kind):
    """Retain literal key/membership/body/store/return subtrees unchanged."""
    original = copy.deepcopy(original_ast)
    body = original.body
    _need(_LEN(body) == 4 and _TYPE(body[0]) is ast.Expr and
          _TYPE(body[1]) is ast.Assign and _TYPE(body[2]) is ast.If and
          _TYPE(body[3]) is ast.Return, "Cache contract original method AST shape changed")
    original_dump = ast.dump(original, include_attributes=False)
    operands = "(lab,)" if kind == "label" else "(small, big)"
    template = ast.parse('''
def _factory(_h25_cache_owner, _h25_cache_error):
    def _method(self):
        _h25_cache_checks = _h25_cache_owner.enter(self)
        try:
            _H25_KEY
            _h25_cache_ticket = _h25_cache_owner.prepare(self, key, OPERANDS, KIND)
            _H25_IF
            _h25_cache_result = _H25_RETURN
            return _h25_cache_owner.finish(self, key, OPERANDS, KIND,
                                          _h25_cache_ticket, _h25_cache_result)
        except _h25_cache_error:
            _h25_cache_owner.abort(_h25_cache_checks)
            raise
    return _method
''')
    factory = template.body[0]
    method = factory.body[0]
    method.name, method.args = original.name, original.args
    method.body.insert(0, body[0])
    factory.body[1].value.id = original.name
    guarded = method.body[2]
    guarded.body[0] = body[1]
    original_if = body[2]
    original_if.test = ast.BoolOp(op=ast.Or(), values=[original_if.test,
        ast.UnaryOp(op=ast.Not(), operand=ast.parse(
            "_h25_cache_owner.reuse(_h25_cache_ticket, key)", mode="eval").body)])
    guarded.body[2] = original_if
    guarded.body[3].value = body[3].value

    class Replace(ast.NodeTransformer):
        def visit_Name(self, node):
            if node.id == "OPERANDS":
                return ast.copy_location(ast.parse(operands, mode="eval").body, node)
            if node.id == "KIND":
                return ast.copy_location(ast.Constant(value=kind), node)
            return node

    Replace().visit(template)
    ast.fix_missing_locations(template)
    compiled = _COMPILE(template, "<h25-cache-contract-v8>", "exec", dont_inherit=True,
                        optimize=sys.flags.optimize)
    # FunctionType avoids adding even temporary globals to the scientific module.
    make = FunctionType(_code_named(compiled, "_factory"), module.__dict__)
    generated = make(owner, _BASE_ERROR)
    _need(ast.dump(original_ast, include_attributes=False) == original_dump,
          "Cache contract generator mutated the original AST")
    return generated


def _bootstrap_api(bootstrap_root, bootstrap_capability, bootstrap_api):
    """Use the independently authenticated actual parent root and capability.

    The parent establishes root identity before this adapter is loaded. This is
    not admission of arbitrary duck-typed root callbacks. All provider source
    reads, hashing, compile and execution belong exclusively to that root.
    """
    _need(_TYPE(PROBE_PROVIDER_SHA256) is _STR and _LEN(PROBE_PROVIDER_SHA256) == 64,
          "Cache probe provider source pin is not configured")
    _need(_TYPE(BOOTSTRAP_ROOT_SHA256) is _STR and _LEN(BOOTSTRAP_ROOT_SHA256) == 64,
          "Cache trusted bootstrap root pin is not configured")
    _need(_TYPE(bootstrap_root) is ModuleType, "Cache requires the actual trusted parent bootstrap root")
    namespace = bootstrap_root.__dict__
    _need(_TYPE(bootstrap_api) is _TUPLE and _LEN(bootstrap_api) == 4 and
          not _ANY(_TYPE(function) is not FunctionType or function.__globals__ is not namespace
                   for function in bootstrap_api),
          "Cache bootstrap entries are not the parent's sealed source-bound API")
    # The parent sealed these closures before driver/scientific code. Their
    # captured pristine guard runs before mutable root helper lookup; no late
    # capture of public root validate/load attributes occurs here.
    identity, validate, load, diagnostics = bootstrap_api
    root_digest = identity()
    _need(_TYPE(root_digest) is _STR and root_digest == BOOTSTRAP_ROOT_SHA256,
          "Cache trusted bootstrap root source differs")
    provider = validate(bootstrap_capability, PROBE_PROVIDER_SHA256)
    _need(_TYPE(provider) is ModuleType, "Cache bootstrap provider type differs")
    return provider, validate, load, identity


def _load_probe(extension_path, expected_extension_sha256, bootstrap_root, bootstrap_capability,
                bootstrap_api):
    """The trusted root owns cold native admission; every load is independent."""
    provider, validate, load, identity = _bootstrap_api(bootstrap_root, bootstrap_capability, bootstrap_api)
    binding = load(bootstrap_capability, extension_path, expected_extension_sha256)
    try:
        _need(validate(bootstrap_capability, PROBE_PROVIDER_SHA256) is provider,
              "Cache bootstrap provider changed during loading")
        _need(_TYPE(binding) is provider.ProbeBinding, "Cache probe binding type differs")
        provider.validate_binding(binding)
    except _BASE_ERROR as primary_error:
        try:
            binding.close()
        except _BASE_ERROR as close_error:
            raise primary_error from close_error
        raise
    return provider, binding, validate, load, identity

def _constant_equal(actual, expected):
    """No custom constant equality; clear actual roots on every error path."""
    try:
        if _TYPE(actual) is not _TYPE(expected):
            return False
        kind = _TYPE(actual)
        if actual is None or actual is Ellipsis:
            return actual is expected
        if kind is _INT or kind is _BOOL or kind is _STR or kind is bytes:
            return actual == expected
        if kind is _TUPLE:
            if _LEN(actual) != _LEN(expected):
                return False
            for index in range(_LEN(actual)):
                if not _constant_equal(actual[index], expected[index]):
                    return False
            return True
        if kind is CodeType:
            return _code_equal(actual, expected)
        return False
    finally:
        actual = expected = None
        kind = None


def _code_equal(actual, expected):
    """Compare every executable/code metadata field through exact primitives."""
    try:
        if _TYPE(actual) is not CodeType or _TYPE(expected) is not CodeType:
            return False
        for field in ("co_argcount", "co_posonlyargcount", "co_kwonlyargcount",
                      "co_nlocals", "co_stacksize", "co_flags", "co_code",
                      "co_names", "co_varnames", "co_freevars", "co_cellvars",
                      "co_exceptiontable", "co_linetable", "co_firstlineno",
                      "co_filename", "co_name", "co_qualname", "co_consts"):
            if not _constant_equal(_OBJGET(actual, field), _OBJGET(expected, field)):
                return False
        return True
    finally:
        actual = expected = None

def _epoch_suite_retire_preserving(self,reason,primary=None,error_channel=None):
    channel=self.emergency_retirement_channel if error_channel is None else error_channel
    _epoch_need(channel['started'] is False and channel['acknowledged'] is False,'Epoch retirement channel already consumed')
    channel['started']=True
    if primary is not None and self.primary_error is None:self.primary_error=primary
    try:self.retire(reason)
    except BaseException as cleanup_error:
        channel['error']=cleanup_error  # fixed raw owner before any optional growth
        self.uncertain=True
        if primary is None:
            if self.primary_error is None:self.primary_error=cleanup_error
        elif cleanup_error is not primary:
            if self.secondary_error is None:self.secondary_error=cleanup_error
            try:self.secondary.append(cleanup_error)
            except BaseException as journal_error:
                channel['journal_error']=journal_error
        # A failed retirement cannot leave private invocation authority active.

        self.active=False;self.terminal=True;self.pending=None

        if primary is None:raise cleanup_error
    return None
