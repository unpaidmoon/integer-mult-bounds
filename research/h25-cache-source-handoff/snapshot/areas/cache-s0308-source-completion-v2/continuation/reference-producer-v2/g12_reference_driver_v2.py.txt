def need(condition, message):
    if not condition:
        raise RuntimeError(message)


def sha(data):
    return hashlib.sha256(data).hexdigest()


def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"), ensure_ascii=True).encode("ascii")


def cpu():
    own = resource.getrusage(resource.RUSAGE_SELF)
    children = resource.getrusage(resource.RUSAGE_CHILDREN)
    return own.ru_utime + own.ru_stime + children.ru_utime + children.ru_stime


def inventory(root, pins, *, exact=True):
    need(root.is_dir() and not root.is_symlink(), "Missing or symlinked input root")
    files = {}
    for path in sorted(root.rglob("*")):
        need(not path.is_symlink(), "Symlink in frozen inputs: " + str(path))
        if path.is_file():
            relative = str(path.relative_to(root))
            if relative not in pins:
                need(not exact, "Undeclared frozen input: " + relative)
                continue
            raw = path.read_bytes()
            expected = pins[relative][0] if type(pins[relative]) is list else pins[relative]
            need(sha(raw) == expected, "Frozen input digest changed: " + relative)
            files[relative] = {"sha256": sha(raw), "mode": stat.S_IMODE(path.stat().st_mode), "bytes": len(raw)}
    need(set(files) == set(pins), "Frozen input inventory incomplete")
    return files


def typed(value):
    """Lossless full return representation, including order and exact types."""
    kind = type(value)
    if kind is Fraction:
        return ["Fraction", value.numerator, value.denominator]
    if value is None:
        return ["None"]
    if kind is bool:
        return ["bool", value]
    if kind is int:
        return ["int", str(value)]
    if kind is float:
        return ["float", value.hex()]
    if kind is str:
        return ["str", value]
    if kind is list or kind is tuple:
        return [kind.__name__, [typed(item) for item in value]]
    if kind is dict:
        return ["dict", [[typed(key), typed(item)] for key, item in value.items()]]
    raise TypeError("Unsupported exact full-certificate return type: " + kind.__name__)


class Phases:
    """Observational wrappers delegate each call and preserve its return."""
    def __init__(self):
        self.rows = {}

    def call(self, name, function, *args, **kwargs):
        wall, used = time.perf_counter(), cpu()
        try:
            return _g12_reference_after_phase(self, name, function, args, kwargs, function(*args, **kwargs))
        finally:
            row = self.rows.setdefault(name, {"calls": 0, "wall_seconds": 0.0, "cpu_seconds_with_children": 0.0})
            row["calls"] += 1
            row["wall_seconds"] += time.perf_counter() - wall
            row["cpu_seconds_with_children"] += cpu() - used

    def wrap(self, owner, attribute, name):
        original = getattr(owner, attribute)
        def observed(*args, **kwargs):
            return self.call(name, original, *args, **kwargs)
        setattr(owner, attribute, observed)
        return original, observed

def _g12_primitive(value):
    """Lossless tagged exact builtin structures; never invoke foreign equality."""
    kind = type(value)
    if kind is Fraction:
        return ['Fraction', str(value.numerator), str(value.denominator)]
    if value is None:
        return ['None']
    if kind is bool:
        return ['bool', value]
    if kind is int:
        return ['int', str(value)]
    if kind is str:
        return ['str', value]
    if kind is bytes:
        return ['bytes', value.hex()]
    if kind is float:
        return ['float', value.hex()]
    if kind is tuple or kind is list:
        return [kind.__name__, [_g12_primitive(item) for item in value]]
    if kind is dict:
        return ['dict', [[_g12_primitive(key), _g12_primitive(item)]
                         for key, item in value.items()]]
    raise TypeError('G12 common projection requires exact original primitive types')


def _g12_bytes(value):
    return canonical(_g12_primitive(value))


def _g12_code_snapshot(code):
    # Reject custom key/equality domains before inspecting the finite code map.
    need(type(code) is dict and all(type(key) is str for key in code),
         'G12 actual original code must be an exact string-keyed dict')
    need(set(code) == {'roles', 'gates', 'sources', 'outputs'},
         'G12 original code field set changed')
    need(type(code['roles']) is int and code['roles'] > 0 and
         type(code['gates']) is list and type(code['sources']) is dict and
         type(code['outputs']) is dict, 'G12 original code containers changed')
    for gate in code['gates']:
        need(type(gate) is tuple and len(gate) == 3 and type(gate[0]) is int and
             type(gate[1]) is tuple and type(gate[2]) is tuple and
             all(type(slot) is int for slot in gate[1] + gate[2]),
             'G12 original gate primitive types changed')
    for source, slot in code['sources'].items():
        need(type(source) is tuple and len(source) == 3 and
             all(type(point) is int for point in source) and type(slot) is int,
             'G12 original source primitive types changed')
    need(all(type(index) is int and type(slot) is int
             for index, slot in code['outputs'].items()),
         'G12 original output primitive types changed')
    return _g12_bytes(code)


def _g12_remaining_count(code):
    # Source-derived count after successful unchanged optimize0 role function;
    # this is not a per-assert execution trace.
    _g12_code_snapshot(code)
    return (2 * sum(len(set(ins + outs)) for _, ins, outs in code['gates']) +
            len(code['outputs']) + code['roles'] + len(code['sources']))

         type(suffix_phases) is tuple and
         all(type(name) is str for name in suffix_phases) and
         suffix_phases == _G12_SUFFIX_PHASES,
         'G12 actual original suffix outputs/phase completions changed')
    return _g12_bytes((_G12_COMMON_SCHEMA, _G12_SCIENTIFIC_PINS, 25, (0, 2, 1),

                       stdout_text, suffix_phases))

def _g12_source_code_shape(code):
    need(type(code) is _G12_STATE['code_type'],
         'G12 source comparison requires exact CodeType')
    def constant(value):
        kind = type(value)
        if value is None or value is Ellipsis:
            return ('singleton', value is Ellipsis)
        if kind is int or kind is bool or kind is str or kind is bytes:
            return (kind.__name__, value)
        if kind is float:
            return ('float', value.hex())
        if kind is tuple:
            return ('tuple', tuple(constant(item) for item in value))
        if kind is frozenset:
            items = [constant(item) for item in value]
            return ('frozenset', tuple(sorted(items)))
        if kind is _G12_STATE['code_type']:
            return ('code', _g12_source_code_shape(value))
        raise RuntimeError('G12 source code has an unsupported constant')
    return tuple(constant(value) for value in (
        code.co_argcount, code.co_posonlyargcount, code.co_kwonlyargcount,
        code.co_nlocals, code.co_stacksize, code.co_flags, code.co_code,
        code.co_names, code.co_varnames, code.co_freevars, code.co_cellvars,
        code.co_exceptiontable, code.co_linetable, code.co_firstlineno,
        code.co_filename, code.co_name, code.co_qualname, code.co_consts))


def _g12_nested(code, name):
    found = [item for item in code.co_consts
             if type(item) is _G12_STATE['code_type'] and item.co_name == name]
    need(len(found) == 1, 'G12 fixed source code path is not unique: ' + name)
    return found[0]


def _g12_names(code):
    pending, names = [code], []
    while pending:
        current = pending.pop()
        for name in current.co_names:
            if name not in names:
                names.append(name)
        pending.extend(item for item in current.co_consts
                       if type(item) is _G12_STATE['code_type'])
    return tuple(names)


def _g12_function_record(function):
    root = _G12_STATE['root']
    need(type(function) is root.FunctionType, 'G12 original callable exact type changed')
    values = tuple(getter(function, root.FunctionType)
                   for _, getter in _G12_STATE['function_fields'])
    code, namespace, builtins_owner, defaults, keywords, closure, attributes = values
    need(type(namespace) is dict and type(builtins_owner) is dict and
         all(type(name) is str for name in namespace) and
         all(type(name) is str for name in builtins_owner) and
         type(attributes) is dict and all(type(name) is str for name in attributes),
         'G12 original callable namespace domain changed')
    keyword_items = () if keywords is None else tuple(keywords.items())
    cells = () if closure is None else tuple((cell, cell.cell_contents) for cell in closure)
    lookups = tuple((name, namespace.get(name, _G12_MISSING),
                    builtins_owner.get(name, _G12_MISSING)) for name in _g12_names(code))
    return function, values, keyword_items, cells, lookups, tuple(attributes.items())

            return slot['original'] if slot['restored'] else slot['observed']

def _g12_cold_call(operation, *args, **kwargs):
    # The original caller-admission sequence is performed before dispatch.
    # Descriptor came only from the actual pre-exposure Root/cold bootstrap.
    descriptor = _G12_STATE['descriptor']
    primitive = descriptor[5]
    type0, function_type, id0 = primitive[0], primitive[1], primitive[2]
    error0, value_error0, zip0 = primitive[7], primitive[8], primitive[18]
    dispatch, record = descriptor[2], descriptor[3]
    if (_G12_STATE['authority'] is not descriptor[1] or dispatch is not record[0] or
            type0(dispatch) is not function_type):
        raise error0('G12 cold caller dispatcher identity differs')
    for (_, getter), expected in zip0(primitive[4], record[1:7]):
        if id0(getter(dispatch, function_type)) != id0(expected):
            raise error0('G12 cold caller dispatcher environment differs')
    if record[4] is not None or record[5] is not None or record[7]:
        raise error0('G12 cold caller dispatcher defaults contract differs')
    for cell, expected in record[8]:
        try:
            observed_id = id0(primitive[6](cell, primitive[5]))
        except value_error0:
            raise error0('G12 cold caller dispatcher cell became empty') from None
        if observed_id != id0(expected):
            raise error0('G12 cold caller dispatcher cell differs')
    return dispatch(operation, *args, **kwargs)

def _g12_authenticate_function(function, source_code, namespace, defaults):
    root = _G12_STATE['root']
    need(type(function) is root.FunctionType and function.__globals__ is namespace and
         function.__builtins__ is _G12_STATE['builtin_owner'] and
         function.__kwdefaults__ is None and function.__closure__ is None and
         type(function.__dict__) is dict and not function.__dict__,
         'G12 source callable owner/default/closure shape changed')
    if defaults is None:
        need(function.__defaults__ is None, 'G12 source callable defaults changed')
    else:
        need(type(function.__defaults__) is tuple and len(function.__defaults__) == len(defaults),
             'G12 source callable default tuple changed')
        for actual, expected in zip(function.__defaults__, defaults):
            need(type(actual) is type(expected) and actual == expected,
                 'G12 source callable literal default changed')
    need(_g12_source_code_shape(function.__code__) == _g12_source_code_shape(source_code),
         'G12 actual callable code differs from pinned source')
    return _g12_function_record(function)


def _g12_source_tree(path, pin):
    root, api = _G12_STATE['root'], _G12_STATE['api']
    api[0]()
    observed, _, source = root._hash_regular(str(path), 1048576, True)
    need(observed == pin and type(source) is bytes, 'G12 literal source bytes changed')
    api[0]()
    code = root._COMPILE(source, str(path), 'exec', dont_inherit=True, optimize=0)
    api[0]()
    return code

def _g12_install_record(tag, owner, attribute, original, observed, code, namespace,
                         closure_values, defaults=None):
    need(len(_G12_STATE['slots']) < len(_G12_SLOT_ORDER) and
         tag == _G12_SLOT_ORDER[len(_G12_STATE['slots'])], 'G12 observer install order changed')
    need(_g12_slot_owner_value(owner, attribute) is observed and
         type(observed) is _G12_STATE['root'].FunctionType and
         observed.__globals__ is namespace and observed.__builtins__ is _G12_STATE['builtin_owner'] and
         observed.__kwdefaults__ is None and not observed.__dict__ and
         _g12_source_code_shape(observed.__code__) == _g12_source_code_shape(code),
         'G12 actual observer provenance changed: ' + tag)
    if defaults is None:
        need(observed.__defaults__ is None, 'G12 observer defaults changed')
    else:
        need(type(observed.__defaults__) is tuple and len(observed.__defaults__) == len(defaults) and
             all(actual is expected for actual, expected in zip(observed.__defaults__, defaults)),
             'G12 observer default identity changed')
    names = observed.__code__.co_freevars
    need(type(observed.__closure__) is tuple and len(observed.__closure__) == len(names) and
         set(names) == set(closure_values), 'G12 observer freevar set changed')
    for name, cell in zip(names, observed.__closure__):
        need(cell.cell_contents is closure_values[name], 'G12 observer closure provenance changed: ' + name)
    _G12_STATE['slots'].append({'tag': tag, 'owner': owner.__dict__ if type(owner) is _G12_STATE['root'].ModuleType else owner,
                              'setter_owner': owner, 'attribute': attribute, 'original': original,
                              'observed': observed, 'record': _g12_function_record(observed), 'restored': False})

             set(values) == {'basis', 'dim', 'key'} and type(label.basis) is dict and
             all(type(key) is int and type(value) is int for key, value in label.basis.items()) and
             type(label.dim) is int and label.dim == len(label.basis) and type(label.key) is tuple and
             all(type(value) is int for value in label.key) and
             label.key == tuple(sorted(label.basis.values())), 'G12 original Label state changed')
    need(_g12_code_snapshot(code) == _G12_STATE['code_before'], 'G12 original code changed before extra call')


def _g12_reference_after_phase(phases, name, function, args, kwargs, result):
    if name not in ('compile_roles', 'verify_labels', 'complete_original_cli'):
        if _G12_STATE['step'] == 'SUFFIX' and name in _G12_SUFFIX_PHASES:
            _G12_STATE['suffix_phases'].append(name)
        return result
    need(phases is _G12_STATE['phases'], 'G12 fixed original phase owner changed')
    _g12_current()
    if name == 'compile_roles':
        need(_G12_STATE['step'] == 'CLI_ENTER' and not kwargs and len(args) == 1 and
             function is _G12_STATE['slots'][4]['original'], 'G12 original compile checkpoint changed')
        _G12_STATE.update({'c': args[0], 'code': result, 'code_before': _g12_code_snapshot(result),
                           'step': 'COMPILED'})
        return result
    if name == 'verify_labels':
        need(_G12_STATE['step'] == 'COMPILED' and not kwargs and len(args) == 1 and
             function is _G12_STATE['slots'][6]['original'], 'G12 original label checkpoint changed')
        checks = args[0]
        _g12_original_prefix(checks)
        _G12_STATE['checks'] = checks
        _G12_STATE['prefix_end'] = _g12_diagnostic_snapshot()
        extra = checks.target_frame((0, 2, 1))
        need(type(extra) is _G12_STATE['circuit'].Label and
             checks.cache[('perp', (0, 2, 1))] is extra and len(checks.cache) == 85196,
             'G12 actual original extra store changed')
        literal = checks.label_ok(checks.everything)
        need(type(literal) is bool, 'G12 original literal result type changed')
        _G12_STATE['literal'] = literal
        _G12_STATE['suffix_start'] = _g12_diagnostic_snapshot()
        _G12_STATE['suffix_rank_index'] = len(_G12_STATE['rank_calls'])
        _G12_STATE['suffix_rank_prefix'] = _g12_rank_snapshot()
        _G12_STATE['step'] = 'SUFFIX'
        return result
    need(_G12_STATE['step'] == 'SUFFIX' and len(args) == 1 and
         type(kwargs) is dict and tuple(kwargs) == ('run_name',) and
         type(kwargs['run_name']) is str and kwargs['run_name'] == '__main__' and
         function is runpy.run_path, 'G12 original complete CLI checkpoint changed')
    _G12_STATE['suffix_phases'].append(name)
    _g12_capture_completed_cli(result)
    _G12_STATE['step'] = 'MARKED'
    marker = _G12ReferenceCompletion('G12 intentional independent original reference completion')
    _G12_STATE['marker'] = marker
    raise marker


def _g12_capture_completed_cli(namespace):
    module = _G12_STATE['circuit']
    captures = _G12_STATE['captures']
    need(type(namespace) is dict and all(type(key) is str for key in namespace) and
         sys.modules.get('complex_circuit') is module and
         namespace.get('ComplexSideCircuit') is module.ComplexSideCircuit and
         namespace.get('Checks') is module.Checks and namespace.get('compile_roles') is module.compile_roles and
         namespace.get('verify_role_frames') is _G12_STATE['slots'][7]['observed'] and
         namespace.get('serializable') is _G12_STATE['slots'][10]['observed'],
         'G12 original CLI namespace/observer provenance changed')
    need(type(captures) is dict and all(type(key) is str for key in captures) and
         set(captures) == {'side', 'full_return'} and type(captures['side']) is tuple and len(captures['side']) == 4,
         'G12 complete actual original capture set changed')
    c, checks, code, roles = captures['side']
    full_return = captures['full_return']
    need(c is _G12_STATE['c'] and checks is _G12_STATE['checks'] and code is _G12_STATE['code'] and
         type(full_return) is dict and all(type(key) is str for key in full_return) and
         full_return.get('role_frames') is roles and type(full_return.get('upstream_commit')) is str and
         full_return['upstream_commit'] == UPSTREAM,
         'G12 same original generation/role/full return binding changed')
    stdout_text = _G12_STATE['stdout'].getvalue()
    certificate = (_G12_STATE['runtime'] / 'certificates/complex-network.json').read_bytes()
    need(stdout_text == EXPECTED_STDOUT and certificate == _G12_STATE['reference_bytes'],
         'G12 original fresh CLI stdout/certificate differs from original immutable reference')
    for name in _G12_ORIGINAL_PHASES[:-1]:
        need(_G12_STATE['phases'].rows.get(name, {}).get('calls') == 1,
             'G12 original phase omitted/repeated before marker: ' + name)
    need('complete_original_cli' not in _G12_STATE['phases'].rows,
         'G12 original CLI phase finally ran before its actual return hook')
    _g12_check_import_paths()
    index = _G12_STATE['suffix_rank_index']
    ranks = _G12_STATE['rank_calls']
    need(_g12_bytes(tuple(ranks[:index])) == _G12_STATE['suffix_rank_prefix'],
         'G12 retained original rank prefix changed')
    common = _g12_common_outputs(code, _G12_STATE['code_before'], roles, full_return,
                                 certificate, stdout_text, tuple(_G12_STATE['suffix_phases']))
    _G12_STATE.update({'namespace': namespace, 'full_return': full_return, 'role_result': roles,
                       'certificate_bytes': certificate, 'stdout_text': stdout_text,
                       'case_count': _g12_remaining_count(code), 'common': common,
                       'cli_end': _g12_diagnostic_snapshot(),
                       'rank_suffix': _g12_bytes(tuple(ranks[index:]))})


def _g12_check_import_paths():
    runtime, engine = _G12_STATE['runtime'], _G12_STATE['engine']
    for name in _G12_STATE['script_names'] - {'complex_network'}:
        module = sys.modules.get(name)
        expected = engine / 'source/complex_circuit.py' if name == 'complex_circuit' else runtime / 'scripts' / (name + '.py')
        need(type(module) is _G12_STATE['root'].ModuleType and
             type(module.__dict__.get('__file__')) is str and Path(module.__file__).resolve() == expected.resolve(),
             'G12 unpinned original scientific import: ' + name)


def _g12_source_output_custody():
    state = _G12_STATE
    need(inventory(state['frozen'], state['pins']) == state['frozen_before'] and
         inventory(HERE / 'engine', ENGINE_PINS) == state['engine_before'],
         'G12 immutable original source bytes/modes changed')
    need(inventory(state['runtime'], state['pins'], exact=False) == state['runtime_before'] and
         inventory(state['engine'], ENGINE_PINS) == state['copied_engine_before'],
         'G12 original runtime source bytes/modes changed')
    files = {str(path.relative_to(state['runtime'])) for path in state['runtime'].rglob('*') if path.is_file()}
    need(files == set(state['pins']) | {'certificates/complex-network.json'},
         'G12 original runtime output file set changed')
    need(Path(__file__).read_bytes() == state['driver_before'] and
         state['reference'].read_bytes() == state['reference_bytes'],
         'G12 selected reference driver/immutable certificate changed')
    _g12_check_import_paths()


def _g12_restore_and_check(success_marker):
    errors = _G12_STATE['secondary_errors']
    try:
        if _G12_STATE.get('phases') is not None:
            _g12_current()
            need(sys.argv is _G12_STATE['old_argv'], 'G12 original main did not restore sys.argv')
            if success_marker:
                for name in _G12_ORIGINAL_PHASES:
                    need(_G12_STATE['phases'].rows.get(name, {}).get('calls') == 1,
                         'G12 original final phase omitted/repeated: ' + name)
    except BaseException as error:
        errors.append((error, error.__traceback__, 'pre-restoration-currentness'))
    for slot in reversed(_G12_STATE['slots']):
        try:
            _g12_current_function(slot['record'])
            need(not slot['restored'] and
                 _g12_slot_owner_value(slot['setter_owner'], slot['attribute']) is slot['observed'],
                 'G12 refusing to overwrite changed/already-restored observer: ' + slot['tag'])
            setattr(slot['setter_owner'], slot['attribute'], slot['original'])
            slot['restored'] = True
        except BaseException as error:
            errors.append((error, error.__traceback__, 'restore-' + slot['tag']))
    # A missing early handle is not fabricated. Existing handles are checked
    # once each even after a primary failure; any failure blocks reference export.
    for name in ('backend_handle', 'circuit_handle'):
        if name in _G12_STATE:
            try:
                _g12_cold_call(2, _G12_STATE[name])
            except BaseException as error:
                errors.append((error, error.__traceback__, 'final-cold2-' + name))
    if success_marker:
        try:
            _g12_source_output_custody()
        except BaseException as error:
            errors.append((error, error.__traceback__, 'final-original-source-output-custody'))


def _g12_emit_secondary_errors():
    for error, traceback, phase in _G12_STATE['secondary_errors']:
        try:
            sys.stderr.write('G12 secondary reference failure: ' + phase + '\n')
            _g12_traceback.print_exception(type(error), error, traceback, file=sys.stderr)
        except BaseException:
            # Raw exception/traceback objects remain retained in the exact
            # source-owned secondary tuple. Never replace the first primary.
            pass


def _g12_export_reference():
    state = _G12_STATE
    need(state['step'] == 'MARKED' and not state['secondary_errors'] and
         all(slot['restored'] for slot in state['slots']),
         'G12 reference export requires complete source-owned restoration')
    common = state['common']
    envelope = {'schema': 'g12-independent-original-reference-envelope-v1',
                'producer_role': 'independent-original-uncached',
                'selected_driver_sha256': sha(state['driver_before']),
                'original_driver_sha256': _G12_ORIGINAL_DRIVER_PIN,
                'authority_source_sha256': _G12_AUTHORITY_PIN,
                'original_scientific_pins': list(_G12_SCIENTIFIC_PINS),
                'h': 25, 'extra_target': [0, 2, 1],
                'literal_original_type': 'bool', 'literal_original_value': state['literal'],
                'remaining_original_case_count': state['case_count'],
                'common_payload_sha256': sha(common), 'common_payload_hex': common.hex(),
                'scope': 'Actual original N22 literal and P07 common remainder observations only; no original PROFILE_ANCHOR_PASS, FULL_PASS, native qualification or G12 approval.'}
    diagnostic = {'schema': 'g12-independent-original-reference-diagnostics-v1',
                  'producer_role': envelope['producer_role'],
                  'selected_driver_sha256': envelope['selected_driver_sha256'],
                  'common_payload_sha256': envelope['common_payload_sha256'],
                  'prefix_end': state['prefix_end'], 'suffix_start': state['suffix_start'],
                  'cli_end': state['cli_end'], 'ordered_rank_suffix_hex': state['rank_suffix'].hex(),
                  'complement_suffix_limit': 'Only cumulative original start/end receipts; no digest subtraction or suffix-stream claim.',
                  'phases': state['phases'].rows,
                  'completion': 'Exact source-owned intentional marker followed by original finally, exact observer restoration, both cold2 checks and original source/output custody.'}
    # New immutable payloads are produced only after all qualification predicates
    # above. Parent must separately retain/hash/admit these actual output bytes.
    for filename, data in (('g12-reference.common.json', canonical(envelope) + b'\n'),
                           ('g12-reference.diagnostics.json', canonical(diagnostic) + b'\n')):
        with (state['output'] / filename).open('xb') as stream:
            stream.write(data)
    state['step'] = 'EXPORTED'


def _g12_reference_entry():
    try:
        main()
    except BaseException as primary:
        is_marker = primary is _G12_STATE.get('marker') and _G12_STATE['step'] == 'MARKED'
        _G12_STATE['primary'] = (primary, primary.__traceback__, primary.__cause__)
        try:
            _g12_restore_and_check(is_marker)
        except BaseException as secondary:
            _G12_STATE['secondary_errors'].append((secondary, secondary.__traceback__, 'restoration-entry'))
        try:
            _g12_emit_secondary_errors()
        except BaseException as secondary:
            _G12_STATE['secondary_errors'].append((secondary, secondary.__traceback__, 'secondary-evidence-entry'))
        if not is_marker:
            raise
        if _G12_STATE['secondary_errors']:
            first, traceback, _ = _G12_STATE['secondary_errors'][0]
            raise first.with_traceback(traceback) from None
        _g12_export_reference()
        return
    raise RuntimeError('G12 reference reached original full-positive publication without its required marker')


def _g12_reference_bootstrap():
    root, capability, api = bootstrap_root, bootstrap_capability, bootstrap_api
    need(type(api) is tuple and len(api) == 4 and type(root) is root.ModuleType and
         all(type(entry) is root.FunctionType and entry.__globals__ is root.__dict__ for entry in api),
         'G12 reference requires actual parent Root bootstrap values')
    need(api[0]() == _G12_ROOT_PIN and sys.flags.optimize == 0,
         'G12 reference actual Root or original assertions differ')
    _G12_STATE.update({'root': root, 'api': api, 'capability': capability,
                       'code_type': type(main.__code__),
                       'builtin_owner': root.builtins.__dict__,
                       'builtin_items': tuple(root.builtins.__dict__.items()),
                       'builtin_values': dict(root.builtins.__dict__),
                       'function_fields': tuple((name, root.FunctionType.__dict__[name].__get__) for name in
                                                ('__code__', '__globals__', '__builtins__', '__defaults__',
                                                 '__kwdefaults__', '__closure__', '__dict__'))})
    authority_path = __file__.rsplit('/', 2)[0] + '/areas/cache-validity/cold_authority_v7.py'
    api[0]()
    digest, _, source = root._hash_regular(authority_path, 1048576, True)
    need(digest == _G12_AUTHORITY_PIN, 'G12 selected cold source not mounted; historical stage is insufficient')
    authority_module = root.ModuleType('_h25_g12_reference_cold_authority')
    namespace = authority_module.__dict__
    namespace.update({'__file__': authority_path, '__builtins__': root.builtins.__dict__})
    api[0]()
    compiled = root._COMPILE(source, authority_path, 'exec', dont_inherit=True, optimize=root._OPTIMIZE)
    api[0]()
    root._EXEC(compiled, namespace)
    api[0]()
    bundle = namespace['capture_epoch_authority'](root, capability, api)
    authority = bundle[1]
    need(type(authority) is tuple and len(authority) == 7 and
         all(type(entry) is root.FunctionType for entry in authority), 'G12 original cold operations shape changed')
    transport = root.FunctionType.__dict__['__dict__'].__get__(authority[0], root.FunctionType)
    descriptor = transport['_h25_caller_descriptor']
    need(type(descriptor) is tuple and len(descriptor) == 6 and descriptor[1] is authority,
         'G12 initial original cold caller descriptor changed')
    _G12_STATE.update({'authority': authority, 'descriptor': descriptor, 'bundle': bundle})
    api[0]()
    driver_digest, _, driver_source = root._hash_regular(__file__, 1048576, True)
    selected_code = root._COMPILE(driver_source, __file__, 'exec', dont_inherit=True, optimize=0)
    api[0]()
    _g12_authenticate_function(main, _g12_nested(selected_code, 'main'), globals(), None)
    phases_code = _g12_nested(selected_code, 'Phases')
    for name in ('__init__', 'call', 'wrap'):
        _g12_authenticate_function(Phases.__dict__[name], _g12_nested(phases_code, name), globals(), None)
    need(type(Phases) is type and Phases.__bases__ == (object,) and
         Phases.__mro__ == (Phases, object), 'G12 source-declared Phases class bases/MRO differ')
    _G12_STATE.update({'phase_owner': Phases, 'phase_bases': Phases.__bases__,
                       'phase_mro': Phases.__mro__})
    stdlib = {'json': json, 'json_loads': json.loads, 'hashlib': hashlib,
              'hashlib_sha256': hashlib.sha256, 'Path': Path, 'Fraction': Fraction,
              'path_instance_type': type(Path(__file__))}
    _G12_STATE['stdlib'] = stdlib
    _G12_STATE['stdlib_functions'] = tuple(_g12_function_record(function)
                                          for function in (json.loads, json.dumps)
                                          if type(function) is root.FunctionType)
    # The source executor, not a scientific import result, owns this initial
    # selected namespace. The two named own-function/class code paths above
    # additionally bind actual main/Phases to the exact selected source bytes.
    _G12_STATE['driver_functions'] = tuple(_g12_function_record(value)
                                          for value in tuple(globals().values())
                                          if type(value) is root.FunctionType and value.__globals__ is globals())
    _G12_STATE['driver_functions'] += tuple(_g12_function_record(Phases.__dict__[name])
                                           for name in ('__init__', 'call', 'wrap'))
    _G12_STATE['phase_members'] = tuple(Phases.__dict__.items())
    _G12_STATE['driver_items'] = tuple(globals().items())
    _G12_STATE['selected_driver_digest'] = driver_digest


_g12_reference_bootstrap()
