    def make_caller_descriptor(operations, label):

        The caller checks the dispatcher record INLINE before invoking it. This

                      cell_get, error0, value_error0, tuple0, dict0, str0,
                      dict_get, dict_len, dict_iter, next0, dict_clear, missing, zip0)

            # This pretarget block uses captured C primitives only. In particular
            # it does not call need(), fresh_function(), or any other validator.
            if type0(index) is not int0 or index < 0 or index >= len0(operations):
                raise error0(label + ': operation index differs')

                    raise error0(label + ': target exact function differs')
                for (name, getter), expected in zip0(field_getters, record[1:7]):
                    if id0(getter(function, function_type)) != id0(expected):
                        raise error0(label + ': target function environment differs')
                keywords = record[5]
                if keywords is not None:
                    if type0(keywords) is not dict0 or dict_len(keywords) != len0(record[7]):
                        raise error0(label + ': target keyword census differs')
                    iterator = current = None
                    valid_domain = True
                    try:
                        iterator = dict_iter(keywords)
                        while True:
                            current = next0(iterator, missing)
                            if current is missing:
                                break
                            if type0(current) is not str0:
                                valid_domain = False
                                break
                            current = None
                    finally:

def need(condition, message):
    if not condition:
        raise RuntimeError(message)


def string_keys(mapping):
    need(_TYPE(mapping) is _DICT, 'Epoch namespace is not exact dict')
    for key in mapping:
        need(_TYPE(key) is _STR, 'Epoch foreign namespace key')


def names(code):
    pending, result = [code], []
    while pending:
        value = pending.pop()
        need(_TYPE(value) is _CODE, 'Epoch code type')
        for name in value.co_names:
            if name not in result:
                result.append(name)
        pending.extend(x for x in value.co_consts if _TYPE(x) is _CODE)
    return tuple(result)


def function_record(function):
    need(_TYPE(function) is _FUNCTION, 'Epoch exact function required')
    fields = tuple(getter(function, _FUNCTION) for _, getter in _FFIELDS)
    code, namespace, builtin_owner, defaults, keywords, closure, attrs = fields
    string_keys(namespace); string_keys(builtin_owner); string_keys(attrs)
    if defaults is not None:
        need(_TYPE(defaults) is _TUPLE, 'Epoch defaults type')
    if keywords is not None:
        string_keys(keywords)
    cells = () if closure is None else tuple((cell, _CELL(cell, types.CellType)) for cell in closure)
    lookups = tuple((name, _GET(namespace, name, _MISSING),
                     _GET(builtin_owner, name, _MISSING)) for name in names(code))
    return (function, fields, () if keywords is None else tuple(keywords.items()),
            cells, lookups, tuple(attrs.items()))


def check_fields(record):
    function, fields, keywords, cells, lookups, attrs = record
    need(_TYPE(function) is _FUNCTION, 'Epoch function replacement')
    for (_, getter), expected in zip(_FFIELDS, fields):
        need(getter(function, _FUNCTION) is expected, 'Epoch function field changed')
    current_keywords = fields[4]
    if current_keywords is not None:
        string_keys(current_keywords)
        need(len(current_keywords) == len(keywords), 'Epoch keyword census')
        for name, expected in keywords:
            need(_GET(current_keywords, name, _MISSING) is expected, 'Epoch keyword value')
    for cell, expected in cells:
        need(_CELL(cell, types.CellType) is expected, 'Epoch closure cell changed')
    current_attrs = fields[6]
    string_keys(current_attrs)
    need(len(current_attrs) == len(attrs), 'Epoch function attributes census')
    for name, expected in attrs:
        need(_GET(current_attrs, name, _MISSING) is expected, 'Epoch function attribute')


def class_record(cls):
    need(_TYPE(cls) is _TYPE, 'Epoch class metaclass')
    proxy = _CLASS_DICT(cls, _TYPE)
    members = tuple(proxy.items())
    need(all(_TYPE(k) is _STR for k, _ in members), 'Epoch class key domain')
    return (cls, _CLASS_MRO(cls, _TYPE), members)


def graph(functions, classes, canonical_functions=None, canonical_classes=None):
    """Explicit class roots; arbitrary state/tuple cells never imply provenance."""
    functions, classes = list(functions), list(classes)
    seen_f, seen_c, records_f, records_c = {}, {}, [], []
    canonical_functions = {} if canonical_functions is None else canonical_functions
    canonical_classes = {} if canonical_classes is None else canonical_classes
    while functions or classes:
        while classes:
            cls = classes.pop()
            if id(cls) in seen_c:
                continue
            need(len(seen_c) < 2048, 'Epoch class graph bound')
            seen_c[id(cls)] = cls
            rec = canonical_classes.get(id(cls))
            rec = class_record(cls) if rec is None else rec
            need(rec[0] is cls, 'Epoch class canonical identity')
            records_c.append(rec)
            for base in rec[1][1:]:
                if type(base) is type:
                    classes.append(base)
            for _, value in rec[2]:
                if type(value) is _FUNCTION:
                    functions.append(value)
                elif type(value) in (staticmethod, classmethod):
                    functions.append(value.__func__)
                elif type(value) is property:
                    functions.extend(x for x in (value.fget, value.fset, value.fdel) if x is not None)
        if functions:
            function = functions.pop()
            if id(function) in seen_f:
                continue
            need(len(seen_f) < 8192, 'Epoch function graph bound')
            seen_f[id(function)] = function
            rec = canonical_functions.get(id(function))
            rec = function_record(function) if rec is None else rec
            need(rec[0] is function, 'Epoch original function canonical identity')
            records_f.append(rec)
            for _, value in rec[3]:
                if type(value) is _FUNCTION:
                    functions.append(value)
            defaults = rec[1][3]
            if defaults is not None:
                functions.extend(x for x in defaults if type(x) is _FUNCTION)
            functions.extend(x for _, x in rec[2] + rec[5] if type(x) is _FUNCTION)
            functions.extend(x for _, x, _ in rec[4] if type(x) is _FUNCTION)
    return tuple(records_f), tuple(records_c)


class EpochLedger:

    def __init__(self):

    def guard_engine(self):

        need(type(self) is cls and _CLASS_MRO(cls, type) is mro, 'Epoch engine class drift')
        current = _CLASS_DICT(cls, type)
        need(len(current) == len(members), 'Epoch engine member census')

    def capture(self, module, backend, source_bytes, factory_results, authority_graph, missing_token):
        self.guard_engine()
        need(self.phase == 'COLD', 'Epoch original capture is one shot')
        need(type(module) is _MODULE and type(backend) is _MODULE and type(source_bytes) is bytes,
             'Epoch original inputs')
        namespace = module.__dict__; string_keys(namespace)
        need(hashlib.sha256(source_bytes).hexdigest() ==
             'd623617f7c5cb3189580883a8ab30c0bd3832688ae3f1f0c1f881bbbbff8e0c1', 'Epoch original source hash')
        # Caller authenticates backend + each pristine factory BEFORE this call.
        label = backend.guard_module(module)
        checks, circuit = namespace['Checks'], namespace['ComplexSideCircuit']
        need(label is namespace['Label'], 'Epoch original Label identity')
        need(type(factory_results) is tuple and len(factory_results) == 3, 'Epoch actual factory outputs')
        p1, construction, prepared = factory_results
        need(type(prepared) is dict and prepared['owner'].module is module, 'Epoch actual prepared owner')
        compiled = compile(source_bytes, namespace['__file__'], 'exec', dont_inherit=True)
        source_codes = {x.co_name: x for x in compiled.co_consts if type(x) is _CODE}
        for cls, name in ((label, 'Label'), (checks, 'Checks'), (circuit, 'ComplexSideCircuit')):
            expected = {x.co_name: x for x in source_codes[name].co_consts if type(x) is _CODE}
            attrs = _CLASS_DICT(cls, type)

        for origin in self.origins.values():
            for record in origin['authority_classes']:
                if record[0] is cls:return record
        return None

    def projected_member(self, cls, name, expected):
        return self.expected(cls,name,expected)

    def check(self, handle):
        origin=self.origin(handle); namespace=origin['namespace'];string_keys(namespace)
        need(origin['module'].__dict__ is namespace and len(namespace)==len(origin['items']),
             'Epoch source namespace owner/census')
        for name, value in origin['items']:
            need(namespace.get(name,_MISSING) is self.expected(namespace,name,value), 'Epoch module current binding')
        graphs=((origin['functions'],origin['classes']),)+tuple(origin['products'])
        seen=set()
        for functions, classes in graphs:
            for cls,mro,members in classes:
                need(type(cls) is type and _CLASS_MRO(cls,type) is mro, 'Epoch class dispatch')
                proxy=_CLASS_DICT(cls,type);need(len(proxy)==len(members),'Epoch class census')
                for name,value in members:
                    need(proxy.get(name,_MISSING) is self.expected(cls,name,value),'Epoch class current member')
            for rec in functions:
                if id(rec[0]) in seen:continue
                seen.add(id(rec[0]));check_fields(rec)
                fn, fields, keywords, cells, lookups,attrs=rec
                ns,builtin_owner=fields[1:3];string_keys(ns);string_keys(builtin_owner)
                for name,old,old_builtin in lookups:
                    need(ns.get(name,origin['missing']) is self.expected_lookup(rec,ns,name,old), 'Epoch current effective global')
                    need(builtin_owner.get(name,origin['missing']) is old_builtin,'Epoch current effective builtin')
        return None

    def phase_to(self,handle,expected,new):
        origin=self.origin(handle);need(origin['phase']==expected,'Epoch phase transition')
        origin['phase']=new;return None

    def fix_raw_guard(self,handle,raw):
        origin=self.origin(handle)
        need(origin['phase']=='NATIVE_PRODUCTS_PREPARED' and origin['raw_guard'] is None,'Epoch raw guard phase')
        origin['raw_guard']=raw;origin['phase']='GLOBALS_COMMITTED'

    def finish_bind(self,handle,result,owner):
        origin=self.origin(handle)
        need(origin['phase']=='GLOBALS_COMMITTED' and type(result) is dict and
             result.get('runtime_guard') is origin['raw_guard'] and owner.runtime is origin['raw_guard'],
             'Epoch actual raw/final runtime guard')
        self.check(handle)
        result['runtime_guard'].check()
        origin['phase']='FINAL_CAPTURED'

    def fail(self,handle,error):
        origin=self.origins.get(id(handle))
        if origin is not None and origin['handle'] is handle:
            origin['phase']='FAILED';origin['failure']=error
        self.failure=error


# Source supplement to the versioned authority; initial source loading is parent-owned.

def _epoch_closure(function):
    need(type(function) is types.FunctionType, 'Epoch wrapper exact function')
    return {} if function.__closure__ is None else dict(zip(function.__code__.co_freevars,
                (cell.cell_contents for cell in function.__closure__)))


def _epoch_code_constant(code, name):
    values=[x for x in code.co_consts if type(x) is types.CodeType and x.co_name==name]
    need(len(values)==1, 'Epoch exact nested code '+name)
    return values[0]


def _epoch_operations(ledger, legacy, capture_graph0, raw_function0, descriptor_factory,

    pending={}
    entries={}
    def before():

    def owned(handle, digest=None):
        before();record=cold_check0(handle)
        if digest is not None:need(record[5]==digest,'Epoch selected source pin')
        return record

    def suite_guard(handle,suite):

        origin=ledger.origin(handle)
        record=owned(origin['cache_handle'],EPOCH_CACHE_SHA256)
        cls=record[2].get('_EpochSuite')
        need(type(suite) is cls and (origin['suite'] is suite or origin['suite'] is None and origin.get('creating') is True),'Epoch actual suite class/generation')
        need(suite.handle is handle and suite.module is origin['module'] and
             suite.namespace is origin['namespace'] and suite.label is origin['label'] and
             suite.checks_class is origin['checks'] and suite.E0 is origin['slots']['module','echelon'][1] and
             suite.owner is origin['factory_results'][2]['owner'],'Epoch fixed suite source bindings')
        return origin,record

    def product_graph(origin,functions=(),classes=()):
        fg,cg=capture_graph0(tuple(functions),tuple(classes)+(origin['label'],origin['checks'],origin['circuit']))
        return fg,tuple((x[0],x[1],x[3]) for x in cg)

    def graph_fields_only(value):

                need(builtin_owner.get(name,missing) is expected_builtin,'Epoch factory current effective builtin')
        for cls,mro,members in classes:
            need(type(cls) is type and type.__dict__['__mro__'].__get__(cls,type) is mro,'Epoch factory class MRO')
            proxy=type.__dict__['__dict__'].__get__(cls,type)
            need(len(proxy)==len(members),'Epoch factory class census')
            for name,expected in members:need(proxy.get(name,missing) is ledger.projected_member(cls,name,expected),'Epoch factory class source')

    def admit(stage,*args):

                 function.__globals__ is sys.modules['__main__'].__dict__ and type(phase_class) is type and
                 function.__globals__.get('Phases') is phase_class,'Epoch actual root-installed driver identity')
            need(id(invocation_token) not in entries,'Epoch duplicate initial driver invocation')
            fg,cg=capture_graph0((function,),(phase_class,))
            record=next(row for row in fg if row[0] is function)
            entries[id(invocation_token)]=(invocation_token,record,(fg,tuple((x[0],x[1],x[3]) for x in cg)),phase_class)
            return record
        if stage=='load':
            need(len(args)==1,'Epoch original loader arguments')
            backend_handle=args[0];record=owned(backend_handle,BACKEND_SHA256)
            need(record[12]=='backend' and record[3]=='native_backend','Epoch production backend owner')
            need(sys.modules.get('complex_circuit') is None,'Epoch fresh production module required')
            loader=record[2].get('load_source')
            need(type(loader) is types.FunctionType and loader.__globals__ is record[2],'Epoch original loader binding')
            module=legacy[1](backend_handle,'load_source','complex_circuit')
            need(type(module) is types.ModuleType and module.__name__=='complex_circuit','Epoch actual loaded module')
            raw=record[1].verify_sources()

        need(stage=='factories' and len(args)==7,'Epoch original factory arguments')
        ticket,set_handle,construction_handle,cache_handle,cache_args,cache_kwargs,context=args
        item=pending.pop(id(ticket),None)
        need(item is not None and item[0] is ticket,'Epoch one-shot genuine load ticket')

        owned(backend_handle,BACKEND_SHA256);owned(set_handle,EPOCH_SET_SHA256)
        owned(construction_handle,EPOCH_CONSTRUCTION_SHA256);cache_record=owned(cache_handle,EPOCH_CACHE_SHA256)
        need(type(cache_args) is tuple and len(cache_args)==2 and type(cache_kwargs) is dict and
             all(type(k) is str for k in cache_kwargs),'Epoch cache factory transport')
        need(set(cache_kwargs)=={'bootstrap_root','bootstrap_capability','bootstrap_api','factory_environment'} and
             cache_kwargs['bootstrap_root'] is root and cache_kwargs['bootstrap_capability'] is capability and
             cache_kwargs['bootstrap_api'] is api,'Epoch actual original bootstrap/factory context')
        need(type(context) is dict and all(type(k) is str for k in context) and set(context)=={'data_handle','data_module','phase_function','phase_class','phase_owner','driver_function','driver_record','invocation_token','disposition_state'},'Epoch exact source context key domain')
        data_record=owned(context['data_handle'],EPOCH_DATA_SHA256)
        need(context['data_module'] is data_record[1],'Epoch current data adapter owner')
        phase_fn=context['phase_function'];phase_owner=context['phase_owner']
        need(type(phase_fn) is types.FunctionType and type(phase_owner) is context['phase_class'] and
             phase_fn is context['phase_class'].__dict__['wrap'] and
             phase_fn.__globals__ is sys.modules['__main__'].__dict__,'Epoch genuine driver phase factory')
        # Reference code is source authenticated DATA, compiled but never executed.
        reference_path=pathlib.Path(__file__).with_name('reference_scientific_driver.py.txt')
        reference=reference_path.read_bytes()
        need(hashlib.sha256(reference).hexdigest()==EPOCH_REFERENCE_DRIVER_SHA256,'Epoch reference driver source')
        original_code=compile(reference,phase_fn.__code__.co_filename,'exec',dont_inherit=True,optimize=sys.flags.optimize)
        phase_code=_epoch_code_constant(_epoch_code_constant(original_code,'Phases'),'wrap')
        need(backend._code_shape(phase_code)==backend._code_shape(phase_fn.__code__) and
             phase_fn.__defaults__ is None and phase_fn.__kwdefaults__ is None and phase_fn.__closure__ is None,
             'Epoch original phase source/calling contract')
        entry=entries.pop(id(context['invocation_token']),None)
        need(entry is not None and entry[0] is context['invocation_token'] and entry[1] is context['driver_record'] and
             entry[3] is context['phase_class'],'Epoch original consumed driver-entry authority')
        graph_fields_only(entry[2])
        driver_record=context['driver_record'];check_fields(driver_record)
        need(driver_record[0] is context['driver_function'] and driver_record[0].__globals__ is phase_fn.__globals__,
             'Epoch initial driver function record')
        disposition_state=context['disposition_state']
        need(type(disposition_state) is dict and all(type(k) is str for k in disposition_state),'Epoch actual parent disposition state')
        execution=driver_record[0].__globals__['_EXECUTION']
        need(type(execution) is dict and all(type(k) is str for k in execution) and execution.get('state') is disposition_state,
             'Epoch existing parent failure path must own acquisition state')
        record=disposition_state.get('epoch_acquisition')
        need(type(record) is dict and all(type(k) is str for k in record) and
             disposition_state.get('cache_binding',missing) is None and disposition_state.get('authority') is legacy and
             disposition_state.get('backend_handle') is backend_handle and disposition_state.get('enabled') is True and
             type(disposition_state.get('owned_states')) is list and any(x is disposition_state for x in disposition_state['owned_states']) and
             disposition_state.get('cache_teardown_attempted') is False,'Epoch continuously reachable disposition route')
        # The pristine original functions/factories run before their mutations.
        backend.guard_module(module)
        p1=legacy[1](set_handle,'bind_verify_map',module)
        construction=legacy[1](construction_handle,'bind_construction',module)
        prepared=legacy[1](cache_handle,'prepare_epoch_cache_contract',module,*cache_args,
            **cache_kwargs,disposition_state=disposition_state,disposition_record=record)
        need(disposition_state['cache_binding'] is prepared and record['prepared'] is prepared and record['published'] is True,
             'Epoch acquisition handoff preceded graph/capture')
        fg,cg=capture_graph0(tuple(v for v in module.__dict__.values() if type(v) is types.FunctionType),
                            (module.Label,module.Checks,module.ComplexSideCircuit))
        handle=ledger.capture(module,backend,raw,(p1,construction,prepared),(fg,cg),missing)
        record['origin_handle']=handle
        origin=ledger.origin(handle)
        origin.update({'backend_handle':backend_handle,'cache_handle':cache_handle,
            'set_handle':set_handle,'construction_handle':construction_handle,'context':context,
            'data_record':data_record,'phase_record':raw_function0(phase_fn),'driver_record':driver_record,
            'creating':False,'native_pending':None,'driver_graph':entry[2]})
        roots=tuple(v for mapping in (p1,construction) for v in mapping.values() if type(v) is types.FunctionType)
        ff,fc=capture_graph0(roots,(type(p1['owner']),type(construction['owner']),type(prepared['owner'])))
        origin['factory_graph']=(ff,tuple((x[0],x[1],x[3]) for x in fc))
        origin['factory_items']=tuple((mapping,tuple(mapping.items())) for mapping in (p1,construction))
        record['phase']='ORIGIN_ACCEPTED_PARENT_OWNED'
        return handle,p1,construction,prepared

    def derive(handle,context):
        before()
        origin=ledger.origin(handle);cache_record=owned(origin['cache_handle'],EPOCH_CACHE_SHA256)
        need(origin['phase']=='ORIGINAL' and origin['context'] is context and not origin['creating'],
             'Epoch suite creation provenance')
        creator=cache_record[2].get('create_epoch_suite')
        need(type(creator) is types.FunctionType and creator.__globals__ is cache_record[2],'Epoch suite source creator')
        origin['creating']=True
        try:
            p1,construction,prepared=origin['factory_results']
            prepared['owner']._epoch_creator_record=(bundle_cell[0][3],handle,context)
            suite=creator(bundle_cell[0][3],handle,origin['module'],origin['backend'].verify_sources(),
                          prepared,p1,construction,context,origin['generation'])
            need(origin['suite'] is suite,'Epoch multiproduct creator ownership')
            origin['suite_snapshot']=(suite.registry,suite.products,suite.caller_caps,suite.descriptor,
                                       suite.prepared,suite.p1,suite.construction)
            return suite
        except BaseException as error:
            ledger.fail(handle,error);raise
        finally:origin['creating']=False

    def check_factory(origin):
        for mapping,items in origin['factory_items']:
            need(type(mapping) is dict and len(mapping)==len(items),'Epoch original factory output census')
            for key,value in items:need(mapping.get(key,missing) is value,'Epoch factory output binding')
        graph_fields_only(origin['factory_graph'])
        graph_fields_only(origin['driver_graph'])

    def wrap_proof(origin,suite,kind,name,role,function):
        need(type(function) is types.FunctionType,'Epoch wrapper function type')
        owner,first,old,chain=origin['slots'][kind,name]
        p1,construction,prepared=origin['factory_results'];context=origin['context']
        if role=='construction_add':need(function is construction['add'],'Epoch original construction add product')
        elif role=='p1':need(function is p1['verify_map'],'Epoch original P1 product')
        elif role in ('public_label','public_residual'):
            need(function is suite.products[role],'Epoch genuine public source product')
        elif role=='checks_capture':
            maker=origin['data_record'][2]['install_epoch_circuit']
            need(function.__code__ is _epoch_code_constant(maker.__code__,'captured_checks') and
                 function.__globals__ is origin['data_record'][2],'Epoch original checks observer source')
            cells=_epoch_closure(function)
            need(set(cells)=={'original_checks_init','capture_checks'} and cells['original_checks_init'] is old,
                 'Epoch checks observer original constructor')
            method=cells['capture_checks']
            need(type(method) is types.MethodType and method.__self__ is p1['owner'] and
                 method.__func__ is type(p1['owner']).__dict__['capture_checks'],'Epoch P1 capture method')
        elif role=='phase':
            check_fields(origin['phase_record']);phase=context['phase_function'];cells=_epoch_closure(function)
            need(function.__code__ is _epoch_code_constant(phase.__code__,'observed') and
                 function.__globals__ is phase.__globals__ and set(cells)=={'self','name','original'} and
                 cells['self'] is context['phase_owner'] and cells['original'] is old and
                 type(cells['name']) is str,'Epoch genuine original phase product')
            expected={('circuit','__init__'):'side_construction',('checks','__init__'):'checks_initialization',
                      ('module','compile_roles'):'compile_roles',('checks','verify_map'):'verify_map',
                      ('checks','verify_labels'):'verify_labels'}
            need((kind,name) in expected and cells['name']==expected[kind,name],'Epoch phase identity')
        elif role=='roles_observer':
            check_fields(origin['driver_record']);maker=context['driver_function'];cells=_epoch_closure(function)
            need(function.__code__ is _epoch_code_constant(maker.__code__,'observe_roles') and
                 function.__globals__ is maker.__globals__ and
                 set(cells)=={'captures','original_roles','phases'} and cells['original_roles'] is old and
                 cells['phases'] is context['phase_owner'] and type(cells['captures']) is dict,
                 'Epoch actual role observation source')
        else:need(False,'Epoch undeclared existing wrapper role')
        defaults=function.__defaults__
        need(function.__kwdefaults__ is None and (type(defaults) is tuple and len(defaults)==1 and defaults[0] is None if role=='roles_observer' else defaults is None),'Epoch wrapper default contract')
        return old

    def product(handle,suite,action,*args):
        origin,record=suite_guard(handle,suite)
        need(type(action) is str,'Epoch product action type')
        if action=='product':
            role,function=args
            need(origin['creating'] is True or origin['suite'] is suite,'Epoch genuine source-product creator')
            # Complete source factories already verified reconstructive inverses;
            # both actual function and exact proof record must be in that suite.
            if origin.get('suite_fields') is None:origin['suite_fields']=object.__getattribute__(suite,'__dict__')
            proof=suite.proofs.get(id(function))
            need(proof is not None and proof[0] is function and suite.products[role] is function,
                 'Epoch multiproduct proof member')
            check_fields(raw_function0(function))
            return ledger.register_product(handle,suite,role,function,
                       product_graph(origin,(function,),(type(suite),)))
        if action=='install':
            kind,name,role,function=args
            need(suite.products.get(role) is function,'Epoch registered callsite product')
            old=origin['slots'][kind,name][2]
            ledger.authorize(handle,kind,name,old,function,role,product_graph(origin,(function,),(type(suite),)))
            ledger.install(handle,kind,name,function)
            return None
        if action=='existing':
            kind,name,role,function=args;check_factory(origin)
            old=wrap_proof(origin,suite,kind,name,role,function)
            ledger.authorize(handle,kind,name,old,function,role,product_graph(origin,(function,),(type(suite),)))
            ledger.install(handle,kind,name,function)
            return None
        need(action=='construction_init' and not args,'Epoch constructor wrapper action')
        check_factory(origin);p1,construction,prepared=origin['factory_results']
        wrapper=construction['wrap_constructor'](construction['original_init'],p1['owner'].capture_c)
        old=origin['slots']['circuit','__init__'][2]
        ledger.authorize(handle,'circuit','__init__',old,wrapper,'construction_init',product_graph(origin,(wrapper,),(type(suite),)))
        ledger.install(handle,'circuit','__init__',wrapper)
        return wrapper

    def stages(handle,suite,stage,*args):
        origin,record=suite_guard(handle,suite)
        need(type(stage) is str,'Epoch bind stage type')
        if stage=='reconcile':
            need(origin['phase']=='FACTORIES_ADMITTED','Epoch prebind phase')
            # Every nonnative slot was installed through the source-owned API.
            ledger.check(handle);origin['phase']='CALLERS_INSTALLED';return None
        if stage=='prepare':
            need(origin['phase']=='CALLERS_INSTALLED' and len(args)==5,'Epoch native preparation stage')
            module,label,original_functions,complement,gram=args
            need(module is origin['module'] and label is origin['label'] and type(original_functions) is list,
                 'Epoch actual original bind arguments')
            rank_calls,trace,native,row,final_guard=suite.bind_inputs
            owned(origin['backend_handle'],BACKEND_SHA256)
            backend=origin['backend'];check_fields(origin['phase_record']);check_fields(origin['driver_record'])
            # The original trace factory ran after original guard_module and
            # before the original native closures were assigned.
            oldrank=origin['slots']['module','rank'][2];traced=module.rank
            cells=_epoch_closure(traced)
            need(traced.__code__ is _epoch_code_constant(backend.attach_trace.__code__,'traced_rank') and
                 traced.__globals__ is backend.__dict__ and cells['original_rank'] is oldrank and
                 cells['rank_calls'] is rank_calls,'Epoch original rank trace source')
            ledger.observed_transition(handle,'module','rank',oldrank,traced,'rank_trace',product_graph(origin,(traced,)))
            expected_helpers=[dict(origin['items'])[name] for name in backend._HELPER_NAMES]
            expected_helpers.extend(v for name,v in origin['class_map'][id(label)][2] if type(v) is types.FunctionType)
            need(len(original_functions)==len(expected_helpers) and all(a is b for a,b in zip(original_functions,expected_helpers)),
                 'Epoch retained original bind function list')
            records=[]
            for name,fn,role,native_name in (('complement_in',complement,'native_complement','complement'),('nondegenerate',gram,'native_gram','gram_rank')):
                expected_code=_epoch_code_constant(suite.bind_expected_code,name)
                need(backend._code_shape(fn.__code__)==backend._code_shape(expected_code) and
                     fn.__globals__ is backend.__dict__ and fn.__builtins__ is builtins.__dict__ and
                     fn.__defaults__ is None and fn.__kwdefaults__ is None,'Epoch native closure code/calling source')
                values=_epoch_closure(fn);need(values.get('runtime',missing) is None,'Epoch native pending runtime')
                need(values.get('label',label) is label,'Epoch native Label closure')
                entry=native.__dict__[native_name]
                need(type(entry) is types.BuiltinFunctionType and entry.__self__ is native and
                     values.get('native_complement' if name=='complement_in' else 'native_gram') is entry,
                     'Epoch current native entry closure')
                for field,value in values.items():
                    if field=='runtime':continue
                    expected={'label':label,'complement_trace':trace,'rank_calls':rank_calls,'rank_row_intern':row,
                        'original_complement':origin['slots']['label','complement_in'][1],
                        'original_nondegenerate':origin['slots']['label','nondegenerate'][1],
                        'original_dot':dict(origin['items'])['dot'],'original_rank':dict(origin['items'])['rank']}.get(field,missing)
                    if expected is not missing:need(value is expected,'Epoch native captured argument '+field)
                    elif field=='counters':need(type(value) is dict and set(value)=={'complement_native','complement_fallback','gram_native','gram_fallback'} and all(type(v) is int and v==0 for v in value.values()),'Epoch actual initial counters')
                    elif field not in ('native_complement','native_gram'):need(False,'Epoch unexpected native closure field '+field)
                cell=dict(zip(fn.__code__.co_freevars,fn.__closure__))['runtime']
                records.append((name,fn,role,cell,raw_function0(fn)))
            origin['native_pending']=tuple(records);origin['phase']='NATIVE_PRODUCTS_PREPARED';return None
        if stage=='commit':
            need(len(args)==1 and origin['phase']=='NATIVE_PRODUCTS_PREPARED','Epoch global commit stage')
            raw=args[0];need(type(raw) is suite.bind_inputs[4],'Epoch exact actual raw guard allocation')
            for name,fn,role,cell,initial in origin['native_pending']:
                need(cell.cell_contents is raw,'Epoch exact fixed runtime-cell endpoint')
                fields=raw_function0(fn)
                need(len(fields[1])==len(initial[1]) and all(a is b for a,b in zip(fields[1],initial[1])) and len(fields[3])==len(initial[3]),'Epoch native closure fields changed')
                for (oldcell,oldvalue),(newcell,newvalue) in zip(initial[3],fields[3]):
                    need(oldcell is newcell and (newvalue is raw if oldcell is cell else newvalue is oldvalue),'Epoch native cell transition')
                old=origin['slots']['label',name][2]
                ledger.observed_transition(handle,'label',name,old,fn,role,product_graph(origin,(fn,),(type(raw),)))
            ledger.fix_raw_guard(handle,raw)
            old=origin['slots']['module','echelon'][2];hook=suite.products['canonical']
            # Only initialization can observe GLOBALS_COMMITTED; no epoch is armed.
            origin['phase']='NATIVE_PRODUCTS_PREPARED'
            ledger.authorize(handle,'module','echelon',old,hook,'canonical',product_graph(origin,(hook,),(type(suite),)))
            ledger.install(handle,'module','echelon',hook);origin['phase']='GLOBALS_COMMITTED'
            ledger.check(handle);return None
        need(stage=='finish' and len(args)==2,'Epoch final binding stage')
        return ledger.finish_bind(handle,args[0],args[1])

    def seal(handle,suite):
        origin,record=suite_guard(handle,suite)
        need(origin['phase']=='FINAL_CAPTURED','Epoch seal after actual final graph')
        ledger.check(handle);origin['phase']='SEALED';return None

    def check(handle,suite):
        origin,record=suite_guard(handle,suite)
        need(origin['phase'] in ('FACTORIES_ADMITTED','CALLERS_INSTALLED','GLOBALS_COMMITTED','FINAL_CAPTURED','SEALED'),
             'Epoch current binding phase')
        if 'suite_snapshot' in origin:
            values=origin['suite_snapshot']
            need(all(a is b for a,b in zip(values,(suite.registry,suite.products,suite.caller_caps,suite.descriptor,
                            suite.prepared,suite.p1,suite.construction))),'Epoch fixed suite data binding')
            for role,fn in origin.get('private_products',{}).items():
                need(suite.products.get(role) is fn,'Epoch actual product dispatch identity')
        graph_fields_only(origin['driver_graph'])
        ledger.check(handle);return None

    def manifest(handle):

    def load(path, digest, name, kind='module'):
        cold_check()
        need(type0(path) is str0 and path.startswith('/') and type0(digest) is str0 and len0(digest) == 64 and
             type0(name) is str0 and name and type0(kind) is str0 and kind in ('module', 'backend'),
             'Cold authority source-load argument shape differs')

        need((kind == 'backend') == (digest == backend_pin), 'Cold authority original backend classification differs')
        domains = {}

        finally:

        need(observed == digest and type0(source_bytes) is bytes0, 'Cold authority source bytes differ')
        module = module_type(name)
        namespace = module_dict_get(module, module_type)
        namespace.update({'__file__': path, '__builtins__': builtin_namespace})
        # Dataclass/import ownership sees the correct unique slot during exec.
        dict_set(initial_sys_modules, name, module)
        try:
            code = root_compile(source_bytes, path, 'exec', dont_inherit=True, optimize=initial_optimize)
            root_exec(code, namespace)
            record = module_image(module, name, kind, path, digest, source_bytes)

            dict_set(by_namespace, id0(namespace), record)
            if kind == 'backend':

            cold_check(record[0])
        except base_error0:
            # A failed source is retained in the unique slot for evidence and
            # cannot be retried/reclassified as an unobserved clean load.
            raise
        return (module, record[0])
